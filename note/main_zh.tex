\begin{abstract}
Wasserstein 鲁棒检验中的变量选择问题具有精确的有限维表示，并可写成一族仿射函数的最大值。对偶运输计划据此生成全局有效的割。
\end{abstract}


\section{问题表述}\label{sec:problem}

本文研究鲁棒假设检验中的变量选择问题。令 $[D]=\{1,\ldots,D\}$，以正整数 $d$ 表示允许选择的变量数。
给定检测函数 $f:~\mathbb{R}^D\to\mathbb{R}$，遵循文献 \cite{gao2018robust} 的约定，将最坏情形风险定义为


\begin{equation}\label{eq:risk}
R(f)=\sup_{\mathbb{P}_0\in\mathcal{P}_0, \mathbb{P}_1\in\mathcal{P}_1}~
\left\{\mathbb E_{\mathbb P_0}\ell(-f(X))+
       \mathbb E_{\mathbb P_1}\ell(f(X))\right\},
\end{equation}
其中，$\mathcal{P}_k, k=0,1$ 为基于 Wasserstein 距离构造的分布不确定集；生成函数 $\ell(\cdot)$ 非负、非减且为凸函数，并满足 $\ell(0)=1, \lim_{t\to-\infty}\ell(t)=0$。
对于 $k=0,1$，假设第 $k$ 类中有 $n$ 个观测，记为 $\{x_i^{(k)}\}_{i=1}^n$。
将相应的经验分布记为 $\widehat{\mathbb{P}}_k$，构造 Wasserstein 不确定集
\[
\mathcal{P}_k=\Big\{
\mathbb{P}_k:~\mathcal{W}_1(
\widehat{\mathbb{P}}_k, \mathbb{P}_k
)\le \rho_k
\Big\}.
\]
这里，$\mathcal{W}_1(\cdot,\cdot)$ 表示以 $\ell_1$ 范数为运输成本的 $1$-Wasserstein 距离。


对于 $S\subseteq[D]$，定义仅依赖于 $S$ 中坐标的检测函数类：
\[
\mathcal{F}_S = \Big\{
f:~\mathbb{R}^D\to\mathbb{R}, f(x) = g(x[S])\text{，其中 $g:~\mathbb{R}^{|S|}\to\mathbb{R}$ 为某个可测函数}
\Big\}.
\]
对于向量 $x\in\mathbb{R}^D$，以 $x[i]$ 表示其第 $i$ 个坐标，以 $x[S]$ 表示下标属于 $S$ 的坐标组成的向量。
所研究的变量选择问题可写成如下组合优化问题：
\begin{equation}
\min_{S\subseteq[D], |S|\le d}\min_{f\in\mathcal{F}_S}~R(f).
\end{equation}

\section{有限维重构与对偶}\label{sec:value}
\subsection{固定选集的有限维重构}\label{sec:fixed}
固定 $S\subseteq[D]$，记 $V(S)=\min_{f\in\mathcal F_S}R(f)$。
假设 $\rho_0,\rho_1>0$，且 $\ell$ 在 $\mathbb R$ 上取有限值。
沿用文献 \cite[引理~2，式~(7)]{gao2018robust} 所列相关稿件的最不利分布表示，
将两类观测合并编号为 $x_1,\ldots,x_{2n}$，其中
$x_{kn+i}=x_i^{(k)}$。令
\[
c_{ir}^{k,S}=\|x_i^{(k)}[S]-x_r[S]\|_1,
\qquad i\in[n],\ r\in[2n],\ k=0,1.
\]
该引理给出选中坐标上随机检验的如下运输问题：
\begin{equation}\label{eq:finite-transport}
\begin{aligned}
\max_{p_0,p_1,\pi^0,\pi^1\ge0}\quad&
\sum_{r=1}^{2n}\min\{p_{0r},p_{1r}\}\\
\text{约束为}\quad&
\sum_{r=1}^{2n}\pi^k_{ir}=\frac1n,\quad i\in[n],\ k=0,1,\\
&p_{kr}=\sum_{i=1}^n\pi^k_{ir},\quad r\in[2n],\ k=0,1,\\
&\sum_{i=1}^n\sum_{r=1}^{2n}c_{ir}^{k,S}\pi^k_{ir}\le\rho_k,
\quad k=0,1.
\end{aligned}
\end{equation}
其中，$\pi^k_{ir}$ 将第 $k$ 类的质量从 $x_i^{(k)}[S]$ 运至
$x_r[S]$，$p_{kr}$ 为该目标点接收到的质量。以
$\mathcal T_k(S)$ 表示满足式~\eqref{eq:finite-transport} 中第 $k$ 类行和约束
与运输预算约束的非负矩阵集合。列和等式据此确定 $p_k$。

对于第~\ref{sec:problem}~节的生成损失，记
$\ell_0(s)=\ell(-s)$、$\ell_1(s)=\ell(s)$。目标点 $r$ 处的最优加权损失为
$\inf_s\{p_{0r}\ell(-s)+p_{1r}\ell(s)\}$。
相应的最不利分布表示推广为
\begin{equation*}
\begin{aligned}
V(S)
&=\max_{\pi^0\in\mathcal T_0(S),\,\pi^1\in\mathcal T_1(S)}
\sum_{r=1}^{2n}\inf_{s\in\mathbb R}
\{p_{0r}\ell(-s)+p_{1r}\ell(s)\}\\
&=\inf_{u\in\mathbb R^{2n}}\sum_{k=0}^1
\max_{\pi^k\in\mathcal T_k(S)}
\sum_{i=1}^n\sum_{r=1}^{2n}\pi^k_{ir}\ell_k(u_r).
\end{aligned}
\end{equation*}
第一个等式需要将该引理推广至一般损失；附录~\ref{app:finite}
通过分别构造检测函数与运输计划，证明两个方向的不等式。
第二个等式来自有限维极小极大对偶。
其中的期望损失项直接由列和得到：
$\sum_r p_{kr}\ell_k(u_r)=\sum_{i,r}\pi^k_{ir}\ell_k(u_r)$。

固定 $u$，对第 $k$ 类的运输线性规划取对偶。
以 $a_i^k\in\mathbb R$ 对应行和约束，以 $\lambda_k\ge0$ 对应运输预算，
其拉格朗日函数为
\[
\frac1n\sum_i a_i^k+\rho_k\lambda_k
+\sum_{i,r}\pi^k_{ir}
\bigl[\ell_k(u_r)-a_i^k-\lambda_kc_{ir}^{k,S}\bigr].
\]
其关于 $\pi^k\ge0$ 的上确界有限，当且仅当所有方括号内的系数均非正。
由线性规划强对偶，再关于 $u$ 最小化，得到
\begin{equation}\label{eq:finite}
\begin{aligned}
V(S)=\min_{a,u,\lambda}\quad&
\underbrace{\frac1n\sum_{k=0}^1\sum_{i=1}^n a_i^k
+\sum_{k=0}^1\rho_k\lambda_k}_{J(a,\lambda)}\\
\text{约束为}\quad&
\ell_k(u_r)\le a_i^k+\lambda_kc_{ir}^{k,S},
\quad i\in[n],\ r\in[2n],\ k=0,1,\\
&a\in\mathbb R^{2\times n},\quad u\in\mathbb R^{2n},\quad
\lambda\in\mathbb R_+^2.
\end{aligned}
\end{equation}
因此，式~\eqref{eq:finite} 的最优值恰好等于仅使用 $S$ 的检测函数所能达到的
最优风险；再对满足 $|S|\le d$ 的选集最小化，即可求解
第~\ref{sec:problem}~节的问题。附录~\ref{app:finite} 的证明还在
合并样本位置之外构造了最优检测函数。

\subsection{混合整数重构}\label{sec:selection}
令 $z=\mathbf1_S$，定义各坐标的运输成本
\[
\Delta^k_{ir,j}=|x_i^{(k)}[j]-x_r[j]|,
\qquad c_{ir}^{k,S}=\sum_{j\in S}\Delta^k_{ir,j}.
\]
按照 Bertsimas 等人 \cite{bertsimas2021unified} 的逻辑约束建模思路，
引入在 $z_j=0$ 时取零的坐标价格 $w_{kj}$。
对于固定的 $z\in[0,1]^D$，定义
\begin{equation}\label{eq:value}
\begin{aligned}
F(z)=\min_{a,u,\lambda,w}\quad&J(a,\lambda)\\
\text{约束为}\quad&
\ell_k(u_r)\le a_i^k+\sum_{j=1}^D\Delta^k_{ir,j}w_{kj},
\quad i\in[n],\ r\in[2n],\ k=0,1,\\
&0\le w_{kj}\le\lambda_k,\quad k=0,1,\ j\in[D],\\
&\rho_0w_{0j}+\rho_1w_{1j}\le z_jJ(a,\lambda),\quad j\in[D],\\
&a\in\mathbb R^{2\times n},\quad u\in\mathbb R^{2n},\quad
\lambda\in\mathbb R_+^2.
\end{aligned}
\end{equation}
选择源点自身作为目标点，得到 $a_i^k\ge0$。
因此，当 $z_j=1$ 时，对应的选择不等式不增加限制：
\[
\rho_0w_{0j}+\rho_1w_{1j}
\le\rho_0\lambda_0+\rho_1\lambda_1\le J(a,\lambda).
\]
当 $z_j=0$ 时，该不等式迫使 $w_{0j}=w_{1j}=0$。
式~\eqref{eq:finite} 的任意可行解都可通过 $w_{kj}=\lambda_kz_j$
扩充为式~\eqref{eq:value} 的可行解。反之，在 $z=\mathbf1_S$ 时，有
\[
\ell_k(u_r)\le a_i^k+\sum_{j\in S}\Delta^k_{ir,j}w_{kj}
\le a_i^k+\lambda_kc_{ir}^{k,S}.
\]
两个方向均保持 $J$ 不变，因此 $F(\mathbf1_S)=V(S)$。
这就证明了该重构对二元选集的精确性。

\subsection{对偶表示}\label{sec:affine}
对式~\eqref{eq:value} 的损失约束与选择约束，分别引入乘子
$\Pi^k_{ir}\ge0$ 和 $\mu_j\ge0$。对偶目标为
\[
H(\Pi)=\sum_{r=1}^{2n}\inf_{s\in\mathbb R}
\sum_{k=0}^1\sum_{i=1}^n\Pi^k_{ir}\ell_k(s).
\]
这是各目标点处最优加权损失之和，其权重由 $\Pi^0$ 和 $\Pi^1$
的列质量给出。关于 $a,u,\lambda,w$ 最小化拉格朗日函数，得到
\begin{equation}\label{eq:dual-normalized}
\begin{aligned}
F(z)=\max_{\Pi,\alpha,\mu}\quad&H(\Pi)\\
\text{约束为}\quad&(\Pi,\alpha,\mu)\in\mathcal D,\qquad
\alpha+\mu^\top z=1,
\end{aligned}
\end{equation}
其中，$(v)_+=\max\{v,0\}$，且
\begin{equation}\label{eq:dual-domain}
\mathcal D=\left\{(\Pi,\alpha,\mu)\ge0:
\begin{array}{ll}
\displaystyle\sum_{r=1}^{2n}\Pi^k_{ir}=\alpha/n,&i\in[n],\ k=0,1,\\[3pt]
\displaystyle\sum_{j=1}^D
\left(\sum_{i,r}\Pi^k_{ir}\Delta^k_{ir,j}-\rho_k\mu_j\right)_+
\le\alpha\rho_k,&k=0,1
\end{array}\right\}.
\end{equation}
行和等式表明，$\Pi^k$ 是源质量经 $\alpha$ 缩放后的运输计划。
附录~\ref{app:dual} 通过消去 $w_{kj}\le\lambda_k$ 的乘子，推导出
上述正部约束。

集合 $\mathcal D$ 为锥，且对于 $\tau\ge0$，有
$H(\tau\Pi)=\tau H(\Pi)$。对任意归一化三元组，
\[
\max_{\tau\ge0}\{2\sqrt{\tau H(\Pi)}-\tau\}=H(\Pi).
\]
优化这一缩放系数即可消去归一化条件，得到
\begin{equation}\label{eq:dual-affine}
\boxed{F(z)=\max_{(\Pi,\alpha,\mu)\in\mathcal D}
\{2\sqrt{H(\Pi)}-\alpha-\mu^\top z\}.}
\end{equation}
将归一化最优解 $(\bar\Pi,\bar\alpha,\bar\mu)$ 乘以 $F(z)$，
即可得到式~\eqref{eq:dual-affine} 的最优解。
因此，其仿射斜率为 $-F(z)\bar\mu$。
命题~\ref{prop:dual} 证明这一表示及相应的次梯度公式。

\section{切平面算法}\label{sec:algorithm}
仿射表示使我们能够采用 Bertsimas 等人 \cite[第~3.1~节]{bertsimas2021unified}
的外逼近策略：对一个选集求解连续问题，保留其仿射下界，
再用二元线性规划选择下一个选集。
记 $\mathrm{OPT}=\min_{|S|\le d}V(S)$，令 $\mathcal Z_d=\{z\in\{0,1\}^D:\mathbf1^\top z\le d\}$。
在 $z^t$ 处，取式~\eqref{eq:dual-affine} 的最优解
$(\Pi^t,\alpha^t,\mu^t)$，定义
\begin{equation}\label{eq:tangent}
\begin{aligned}
c_t(z)&=2\sqrt{H(\Pi^t)}-\alpha^t-(\mu^t)^\top z,\\
F(z)&\ge c_t(z)=F(z^t)-(\mu^t)^\top(z-z^t),\qquad
c_t(z^t)=F(z^t).
\end{aligned}
\end{equation}
对偶可行性给出下界，对偶最优性给出在 $z^t$ 处的等式。
因此，这是一个支撑仿射函数，而非任意下界；此处不要求 $F$ 可微。
下一选集与主问题下界由以下问题给出
\begin{equation}\label{eq:master}
\begin{aligned}
(z^{t+1},\eta^{t+1})\in
\operatorname*{arg\,min}_{z\in\mathcal Z_d,\ \eta\ge0}\quad&\eta\\
\text{约束为}\quad&
\eta\ge F(z^s)-(\mu^s)^\top(z-z^s),\qquad s=1,\ldots,t.
\end{aligned}
\end{equation}
记 $U_t=\min_{s\le t}F(z^s)$。由于保留的全部割均位于 $F$ 下方，有
\[
\eta^{t+1}\le\mathrm{OPT}\le U_t.
\]
当 $U_t-\eta^{t+1}\le\varepsilon$ 时停止，返回已求值选集中最优的一个。
附录~\ref{app:finite} 说明如何由有限维凸问题的解恢复对应的检测函数。

\begin{algorithm}[H]
\caption{变量选择的精确切平面方法}\label{alg:oa}
\small
\begin{algorithmic}[1]
\Require 问题数据、预算 $d$、容差 $\varepsilon>0$ 及 $z^1\in\mathcal Z_d$。
\Ensure 选集 $S^*$、对应的最优检测函数 $f^*$，以及宽度至多为 $\varepsilon$ 的风险区间。
\State $t\gets1$, $U\gets+\infty$.
\While{真}
  \State 在 $z^t$ 处求解式~\eqref{eq:dual-affine}，取最优解 $(\Pi^t,\alpha^t,\mu^t)$。
  \If{$F(z^t)<U$}
    \State $(U,z^*)\gets(F(z^t),z^t)$.
  \EndIf
  \State 保留下标为 $t$ 的割~\eqref{eq:tangent}。
  \State 求解式~\eqref{eq:master}，得到 $(z^{t+1},\eta^{t+1})$。
  \If{$U-\eta^{t+1}\le\varepsilon$}
    \State $S^*\gets\{j:z_j^*=1\}$.
    \State 对 $S^*$ 求解式~\eqref{eq:finite}，并按式~\eqref{eq:extension} 构造 $g^*$。
    \State 令 $f^*(x)=g^*(x[S^*])$，\Return $(S^*,f^*,[\eta^{t+1},U])$。
  \EndIf
  \State $t\gets t+1$.
\EndWhile
\end{algorithmic}
\end{algorithm}

\section{有限维重构}\label{app:finite}
\begin{proposition}[有限维重构]\label{prop:finite}
对于每个 $S\subseteq[D]$，式~\eqref{eq:finite} 的最小值可以取得，且
\begin{equation}\label{eq:source-finite}
\begin{aligned}
V(S)
&=\min_{u\in\mathbb R^{2n}}\sum_{k=0}^1
\max_{\pi^k\in\mathcal T_k(S)}
\sum_{i=1}^n\sum_{r=1}^{2n}\pi^k_{ir}\ell_k(u_r)\\
&=\max_{\substack{\pi^0\in\mathcal T_0(S)\\
                  \pi^1\in\mathcal T_1(S)}}
\sum_{r=1}^{2n}\inf_{s\in\mathbb R}
\{p_{0r}\ell(-s)+p_{1r}\ell(s)\}.
\end{aligned}
\end{equation}
其中，$p_{kr}=\sum_{i=1}^n\pi^k_{ir}$。
式~\eqref{eq:finite} 的最优解还可在 $\mathbb R^D$ 上构造最优检测函数
$f(x)=g(x[S])$。
\end{proposition}
\begin{proof}
固定 $u$ 与类别 $k$，对运输线性规划的源质量约束引入
对偶变量 $a_i^k\in\mathbb R$，对预算约束引入 $\lambda_k\ge0$。
其拉格朗日函数为
\begin{equation}\label{eq:transport-lagrangian}
\begin{aligned}
L_k={}&\sum_{i,r}\pi^k_{ir}\ell_k(u_r)
+\sum_i a_i^k\left(\frac1n-\sum_r\pi^k_{ir}\right)
+\lambda_k\left(\rho_k-\sum_{i,r}c_{ir}^{k,S}\pi^k_{ir}\right)\\
={}&\frac1n\sum_i a_i^k+\rho_k\lambda_k
+\sum_{i,r}\pi^k_{ir}
 [\ell_k(u_r)-a_i^k-\lambda_kc_{ir}^{k,S}].
\end{aligned}
\end{equation}
其关于 $\pi^k\ge0$ 的上确界有限，当且仅当所有方括号内的系数均非正。
由线性规划对偶，有
\begin{equation}\label{eq:transport-dual}
\begin{aligned}
\max_{\pi^k\in\mathcal T_k(S)}
\sum_{i,r}\pi^k_{ir}\ell_k(u_r)
=\min_{a^k\in\mathbb R^n,\,\lambda_k\ge0}\quad&
\frac1n\sum_i a_i^k+\rho_k\lambda_k\\
\text{约束为}\quad&
\ell_k(u_r)\le a_i^k+\lambda_kc_{ir}^{k,S},
\quad i\in[n],\ r\in[2n].
\end{aligned}
\end{equation}
因此，式~\eqref{eq:finite} 中的下确界 $W$ 等于
式~\eqref{eq:source-finite} 第一行的下确界。

对于任意 $f(x)=g(x[S])$，取 $u_r=g(x_r[S])$。
运输计划 $\pi^k\in\mathcal T_k(S)$ 可构造 $\mathcal P_k$ 中的分布：
将质量 $\pi^k_{ir}$ 从 $x_i^{(k)}$ 运至选中坐标为 $x_r[S]$、
其余坐标仍为 $x_i^{(k)}[S^c]$ 的点。
其运输成本为 $\sum_{i,r}c_{ir}^{k,S}\pi^k_{ir}$，
期望损失为 $\sum_{i,r}\pi^k_{ir}\ell_k(u_r)$。因此，
\[
W\le\sum_{k=0}^1\max_{\pi^k\in\mathcal T_k(S)}
\sum_{i,r}\pi^k_{ir}\ell_k(g(x_r[S]))\le R(f).
\]
对 $f\in\mathcal F_S$ 取下确界，得 $W\le V(S)$。

为证明反向不等式，取式~\eqref{eq:finite} 的可行解 $(a,u,\lambda)$。
对于 $y\in\mathbb R^{|S|}$，定义
\[
\begin{gathered}
E_k(y)=\min_{i\in[n]}
\{a_i^k+\lambda_k\|x_i^{(k)}[S]-y\|_1\},\\
i_k=\min\operatorname*{arg\,min}_{i\in[n]}
\{a_i^k+\lambda_k\|x_i^{(k)}[S]-y\|_1\}.
\end{gathered}
\]
令 $r_k=\|x_{i_k}^{(k)}[S]-y\|_1$、
$b=\|x_{i_0}^{(0)}[S]-x_{i_1}^{(1)}[S]\|_1$。定义
\begin{equation}\label{eq:extension}
g(y)=\begin{cases}
(1-\gamma)u_{i_0}+\gamma u_{n+i_1},\quad
\gamma=\min\{1,r_0/b\},&b>0,\\
u_{i_0},&b=0.
\end{cases}
\end{equation}
当 $b>0$ 时，三角不等式给出
$\gamma b\le r_0$ 和 $(1-\gamma)b\le r_1$。
由凸性与式~\eqref{eq:finite} 的源点--目标点约束，有
\[
\begin{aligned}
\ell_0(g(y))
&\le(1-\gamma)\ell_0(u_{i_0})+\gamma\ell_0(u_{n+i_1})
\le a_{i_0}^0+\lambda_0\gamma b\le E_0(y),\\
\ell_1(g(y))
&\le(1-\gamma)\ell_1(u_{i_0})+\gamma\ell_1(u_{n+i_1})
\le a_{i_1}^1+\lambda_1(1-\gamma)b\le E_1(y).
\end{aligned}
\]
当 $b=0$ 时，目标点 $i_0$ 处的两个约束给出同样的界。
取最小下标的规则使 $g$ 为 Borel 可测，且 $|g(y)|\le\max_r|u_r|$。
对于任意 $\mathbb P_k\in\mathcal P_k$，
取从 $\widehat{\mathbb P}_k$ 到 $\mathbb P_k$、成本至多为 $\rho_k$ 的耦合，
对上述界积分，得到
\[
\mathbb E_{\mathbb P_k}\ell_k(g(X[S]))
\le\frac1n\sum_i a_i^k+\rho_k\lambda_k,
\qquad R(f)\le J(a,\lambda),\quad f(x)=g(x[S]).
\]
因此，$V(S)\le W$，从而证明了与有限维凸问题的等价性。

乘积集合 $\mathcal T_0(S)\times\mathcal T_1(S)$ 为紧凸集。
函数 $\sum_{k,i,r}\pi^k_{ir}\ell_k(u_r)$ 连续，关于运输计划为仿射函数，
关于 $u$ 为凸函数。由有限维极小极大对偶，有
\[
\begin{aligned}
W
&=\inf_u\max_{\pi^0,\pi^1}
  \sum_{k,i,r}\pi^k_{ir}\ell_k(u_r)
 =\max_{\pi^0,\pi^1}\inf_u
  \sum_{k,i,r}\pi^k_{ir}\ell_k(u_r)\\
&=\max_{\pi^0,\pi^1}\sum_r\inf_s
  \{p_{0r}\ell(-s)+p_{1r}\ell(s)\},
\end{aligned}
\]
其中，每对运输计划均取遍同一乘积集合。
这就从式~\eqref{eq:finite-transport} 的运输约束出发，证明了一般损失下的表示。

最后，$(a,u,\lambda)=(\mathbf1,0,0)$ 为可行解，且 $J=2$。
在下水平集 $J\le2$ 上，对角约束给出 $0\le a_i^k\le2n$，
并且 $0\le\lambda_k\le2/\rho_k$。
其余约束同时界住 $\ell(u_r)$ 和 $\ell(-u_r)$。
由于 $t\to\infty$ 时 $\ell(t)\to\infty$，该下水平集为紧集，
故最小值可以取得。恢复的检测函数因而达到 $V(S)$，其有限得分达到
式~\eqref{eq:source-finite} 中的第一个最小值。
最后一个最大值也可以取得，因为其目标在紧运输集合上为上半连续函数。
\end{proof}

\section{对偶性}\label{app:dual}
\begin{proposition}[精确性与对偶表示]\label{prop:dual}
\textup{(i)} 对于每个 $S\subseteq[D]$，有 $F(\mathbf1_S)=V(S)$。
对于每个 $z\in[0,1]^D$，原问题最小值与归一化对偶最大值均可以取得，且
\begin{equation}\label{eq:appendix-normalized}
0<F(z)=\max_{\substack{(\Pi,\alpha,\mu)\in\mathcal D\\
\alpha+\mu^\top z=1}}H(\Pi)\le2.
\end{equation}

\textup{(ii)} 同一最优值还可以在下式中取得：
\begin{equation}\label{eq:appendix-affine}
F(z)=\max_{(\Pi,\alpha,\mu)\in\mathcal D}
\{2\sqrt{H(\Pi)}-\alpha-\mu^\top z\}.
\end{equation}
若 $(\bar\Pi,\bar\alpha,\bar\mu)$ 为归一化最优解，
则式~\eqref{eq:appendix-affine} 的一个最优解为
\begin{equation}\label{eq:dual-rescaling}
(\Pi,\alpha,\mu)=F(z)(\bar\Pi,\bar\alpha,\bar\mu),\qquad
H(\Pi)=F(z)^2.
\end{equation}

\textup{(iii)} 函数 $F$ 在 $[0,1]^D$ 上为凸函数。
式~\eqref{eq:appendix-affine} 在 $z^t$ 处的最优解
$(\Pi^t,\alpha^t,\mu^t)$ 给出
\begin{equation}\label{eq:support-contact}
\begin{aligned}
c_t(z)&=2\sqrt{H(\Pi^t)}-\alpha^t-(\mu^t)^\top z\le F(z),\\
c_t(z^t)&=F(z^t),\qquad
F(z)\ge F(z^t)-(\mu^t)^\top(z-z^t).
\end{aligned}
\end{equation}
因此，$-\mu^t\in\partial F(z^t)$，其中次梯度不等式在 $[0,1]^D$ 上成立。
\end{proposition}
\begin{proof}
\textup{(i)} 在两种原问题的损失约束中，取源点自身作为目标点，
均有 $a_i^k\ge0$。
对于式~\eqref{eq:finite} 的可行解 $(a,u,\lambda)$，
令 $z=\mathbf1_S$、$w_{kj}=\lambda_kz_j$，则
\[
\ell_k(u_r)\le a_i^k+\sum_{j=1}^D\Delta_{ir,j}^kw_{kj},\qquad
\sum_{k=0}^1\rho_kw_{kj}
=z_j\sum_{k=0}^1\rho_k\lambda_k\le z_jJ(a,\lambda).
\]
反之，由式~\eqref{eq:value} 在 $z=\mathbf1_S$ 时的可行性
以及 $\rho_k>0$，有
\[
w_{kj}=0\quad(j\notin S),\qquad
\ell_k(u_r)\le a_i^k+\sum_{j\in S}\Delta_{ir,j}^kw_{kj}
\le a_i^k+\lambda_kc_{ir}^{k,S}.
\]
因此，消去 $w$ 后，两种表示具有相同的可行变量 $(a,u,\lambda)$
和目标函数。这就证明了 $F(\mathbf1_S)=V(S)$。

对式~\eqref{eq:value} 的损失约束、不等式 $w_{kj}\le\lambda_k$
及选择约束，分别引入非负乘子 $\Pi_{ir}^k$、$\beta_{kj}$ 和 $\mu_j$。
保留 $a,u$ 为自由变量、$\lambda,w\ge0$，
记 $\alpha=1-\mu^\top z$。拉格朗日函数为
\begin{equation}\label{eq:lagrangian}
\begin{aligned}
\mathcal L={}&J(a,\lambda)
+\sum_{k,i,r}\Pi_{ir}^k
  \left[\ell_k(u_r)-a_i^k-\sum_j\Delta_{ir,j}^kw_{kj}\right]\\
&+\sum_{k,j}\beta_{kj}(w_{kj}-\lambda_k)
+\sum_j\mu_j\left[\sum_k\rho_kw_{kj}-z_jJ(a,\lambda)\right]\\
={}&\sum_{k,i}a_i^k\left(\frac\alpha n-\sum_r\Pi_{ir}^k\right)
+\sum_{r=1}^{2n}\sum_{k=0}^1\sum_{i=1}^n
  \Pi_{ir}^k\ell_k(u_r)\\
&+\sum_k\lambda_k\left(\alpha\rho_k-\sum_j\beta_{kj}\right)
+\sum_{k,j}w_{kj}
  \left[\beta_{kj}+\rho_k\mu_j-\sum_{i,r}\Pi_{ir}^k\Delta_{ir,j}^k\right].
\end{aligned}
\end{equation}
其关于 $a,\lambda,w$ 的下确界有限，当且仅当
\begin{equation}\label{eq:dual-conditions}
\begin{gathered}
\sum_{r=1}^{2n}\Pi_{ir}^k=\frac\alpha n
\quad(k=0,1,\ i\in[n]),\qquad
\sum_{j=1}^D\beta_{kj}\le\alpha\rho_k\quad(k=0,1),\\
\beta_{kj}\ge
\left(\sum_{i=1}^n\sum_{r=1}^{2n}\Pi_{ir}^k\Delta_{ir,j}^k
-\rho_k\mu_j\right)_+
\quad(k=0,1,\ j\in[D]).
\end{gathered}
\end{equation}
行和等式强制 $\alpha\ge0$。
关于 $u$ 的最小化可按 $2n$ 个目标点分离，得到
\[
\inf_u\sum_{r=1}^{2n}\sum_{k=0}^1\sum_{i=1}^n
\Pi_{ir}^k\ell_k(u_r)
=\sum_{r=1}^{2n}\inf_{s\in\mathbb R}
\sum_{k=0}^1\sum_{i=1}^n\Pi_{ir}^k\ell_k(s)=H(\Pi).
\]
在式~\eqref{eq:dual-conditions} 中消去 $\beta$，得到
\[
\sum_{j=1}^D
\left(\sum_{i=1}^n\sum_{r=1}^{2n}\Pi_{ir}^k\Delta_{ir,j}^k
-\rho_k\mu_j\right)_+\le\alpha\rho_k,\qquad k=0,1.
\]
这些恰好是定义 $\mathcal D$ 的约束；
$\alpha=1-\mu^\top z$ 则给出式~\eqref{eq:appendix-normalized} 的归一化条件。

为在某些 $z_j=0$ 时也证明强对偶，消去对应的变量 $w_{kj}$
及其冗余上界约束与选择约束。
消去后的问题具有如下严格可行点
\[
u=0,\qquad a_i^k=2,\qquad\lambda_k=1,\qquad
w_{kj}=z_j/2\quad(z_j>0).
\]
事实上，$\ell_k(0)=1<2$、$0<w_{kj}<1$，且
\[
\sum_{k=0}^1\rho_kw_{kj}
=\frac{z_j}{2}(\rho_0+\rho_1)
<z_j(4+\rho_0+\rho_1)=z_jJ(a,\lambda)
\quad(z_j>0).
\]
由 Slater 定理，消去后的原问题与对偶问题最优值相等，且对偶最大值可以取得。
对于每个消去的坐标，令
\[
\mu_j=\max_{k=0,1}\frac1{\rho_k}
\sum_{i=1}^n\sum_{r=1}^{2n}\Pi_{ir}^k\Delta_{ir,j}^k,
\qquad\beta_{kj}=0\quad(z_j=0).
\]
即可恢复完整对偶。这不改变目标与归一化条件，并满足
式~\eqref{eq:dual-conditions}。因此，对每个 $z\in[0,1]^D$，
归一化对偶均取得 $F(z)$。

可行点 $a_i^k=1$、$u=0$、$\lambda=w=0$ 给出 $F(z)\le2$。
在下水平集 $J\le2$ 上，非负性给出
$a_i^k\le2n$、$\lambda_k\le2/\rho_k$ 和 $w_{kj}\le2/\rho_k$。
损失约束进而同时界住 $\ell(u_r)$ 和 $\ell(-u_r)$。
由凸性、单调性、$\ell(0)=1$ 以及
$\lim_{t\to-\infty}\ell(t)=0$，可知
$t\to\infty$ 时 $\ell(t)\to\infty$。因此，该下水平集为紧集，
原问题最小值可以取得。若该值为零，则有 $a=\lambda=w=0$，与下式矛盾：
\[
2=2\ell(0)\le\ell(-u_r)+\ell(u_r)\le0.
\]
这就证明了式~\eqref{eq:appendix-normalized}。

\textup{(ii)} 集合 $\mathcal D$ 为锥，且对于 $\tau\ge0$，有
$H(\tau\Pi)=\tau H(\Pi)\ge0$。
对于 $(\Pi,\alpha,\mu)\in\mathcal D$，令
$b=\alpha+\mu^\top z$。若 $b>0$，将该三元组除以 $b$，
再应用 \textup{(i)}，得到 $H(\Pi)/b\le F(z)$。因此，
\begin{equation}\label{eq:scaling}
2\sqrt{H(\Pi)}-b
=\frac{H(\Pi)}b-
\left(\sqrt{\frac{H(\Pi)}b}-\sqrt b\right)^2\le F(z).
\end{equation}
若 $b=0$，则 $\alpha=0$，行和等式给出 $\Pi=0$，
故仿射目标为零。反之，对于归一化最优解
$(\bar\Pi,\bar\alpha,\bar\mu)$，有
\[
H(F(z)\bar\Pi)=F(z)^2,\qquad
2\sqrt{H(F(z)\bar\Pi)}
-F(z)(\bar\alpha+\bar\mu^\top z)=F(z).
\]
因此，该仿射目标的上界可以取得，即得
式~\eqref{eq:appendix-affine} 与式~\eqref{eq:dual-rescaling}。

\textup{(iii)} 可行域 $\mathcal D$ 不依赖于 $z$，
因而式~\eqref{eq:appendix-affine} 将 $F$ 表示为一族仿射函数的最大值，
并证明其凸性。$(\Pi^t,\alpha^t,\mu^t)$ 的可行性给出
对每个 $z\in[0,1]^D$ 均有 $c_t(z)\le F(z)$，
在 $z^t$ 处的最优性则给出 $c_t(z^t)=F(z^t)$。因此，
\[
F(z)\ge c_t(z)=c_t(z^t)-(\mu^t)^\top(z-z^t)
=F(z^t)-(\mu^t)^\top(z-z^t).\qedhere
\]
\end{proof}
