\begin{abstract}
本文从 Wasserstein 鲁棒检验的有限维最不利分布表示出发，推导带坐标选择的凸模型。利用统一的风险上界引入选择变量，保持二元点处的最优值不变。所得凸值函数具有仿射对偶割，可用于在变量选集上优化。
\end{abstract}


\section{问题表述}\label{sec:problem}

本文研究鲁棒假设检验中的变量选择问题。令 $[D]=\{1,\ldots,D\}$，以正整数 $d$ 表示允许选择的变量数。
给定检测函数 $f:~\mathbb{R}^D\to\mathbb{R}$，遵循文献 \cite{gao2018robust} 的约定，将最坏情形风险定义为


\begin{equation}\label{eq:risk}
R(f)=\sup_{\mathbb{P}_0\in\mathcal{P}_0, \mathbb{P}_1\in\mathcal{P}_1}~
\left\{\mathbb E_{\mathbb P_0}\ell(-f(X))+
       \mathbb E_{\mathbb P_1}\ell(f(X))\right\},
\end{equation}
其中，$\mathcal{P}_k, k=0,1$ 为基于 Wasserstein 距离构造的分布不确定集；生成函数 $\ell(\cdot)$ 非负、非减且为凸函数，并满足 $\ell(0)=1, \lim_{t\to-\infty}\ell(t)=0$。
对于 $k=0,1$，假设第 $k$ 类中有 $n_k$ 个观测，记为 $\{x_i^{(k)}\}_{i=1}^{n_k}$。
将相应的经验分布记为 $\widehat{\mathbb{P}}_k=\frac1{n_k}\sum_{i=1}^{n_k}\delta_{x_i^{(k)}}$，构造 Wasserstein 不确定集
\[
\mathcal{P}_k=\Big\{
\mathbb{P}_k:~\mathcal{W}_1(
\widehat{\mathbb{P}}_k, \mathbb{P}_k
)\le \rho_k
\Big\}.
\]
这里，$\mathcal{W}_1(\cdot,\cdot)$ 表示以 $\ell_1$ 范数为运输成本的 $1$-Wasserstein 距离。


对于 $S\subseteq[D]$，定义仅依赖于 $S$ 中坐标的检测函数类：
\[
\mathcal{F}_S = \Big\{
f:~\mathbb{R}^D\to\mathbb{R}, f(x) = g(x[S])\text{，其中 $g:~\mathbb{R}^{|S|}\to\mathbb{R}$ 为某个可测函数}
\Big\}.
\]
对于向量 $x\in\mathbb{R}^D$，以 $x[i]$ 表示其第 $i$ 个坐标，以 $x[S]$ 表示下标属于 $S$ 的坐标组成的向量。
所研究的变量选择问题可写成如下组合优化问题：
\begin{equation}
\min_{S\subseteq[D], |S|\le d}\min_{f\in\mathcal{F}_S}~R(f).
\end{equation}

\section{有限维等价表示}\label{sec:finite}
假设 $n_0,n_1\ge1$、$\rho_0,\rho_1>0$，且 $\ell$ 在 $\mathbb R$ 上有限。
所有分布的一阶矩均有限。固定 $S\subseteq[D]$，记
\[
V(S)=\min_{f\in\mathcal F_S}R(f),\qquad
\mathcal X=\{x_i^{(0)}:i\in[n_0]\}\cup
\{x_j^{(1)}:j\in[n_1]\}.
\]
源观测保留其类别下标，以 $y\in\mathcal X$ 表示目标点。
投影到 $S$ 不会增加运输成本。反过来，保持各源观测的未选坐标不变，
即可将投影后观测的任意运输提升到原空间，且成本不变。
因此，投影后的 Wasserstein 球恰好描述了与检测函数 $g(x[S])$ 有关的不确定性。

以文献 \cite[引理~2，式~(7)]{xie2021robust} 的完整支撑运输表示为起点。
使用本文类别下标后，其随机化检验目标在每个目标点使用 $\min\{p_0(y),p_1(y)\}$。
对于第~\ref{sec:problem}~节的生成损失，这一项变为实值评分上的加权损失下确界。
以下引理保留原论文的运输约束，并沿用本文的类别下标。
附录~\ref{app:recovery} 证明其向当前损失的推广。

\begin{lemma}[有限维最不利分布重构]\label{prop:lfd}
固定选集的风险为
\begin{equation}\label{eq:lfd}
\begin{aligned}
V(S)=\max_{p,\gamma\ge0}\quad&
\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\{p_0(y)\ell(-v)+p_1(y)\ell(v)\}\\
\text{约束为}\quad&
\sum_{y\in\mathcal X}\gamma_{i,y}^{(k)}=\frac1{n_k},
&&i\in[n_k],\ k=0,1,\\
&p_k(y)=\sum_{i=1}^{n_k}\gamma_{i,y}^{(k)},
&&y\in\mathcal X,\ k=0,1,\\
&\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}\|x_i^{(k)}[S]-y[S]\|_1\le\rho_k,
&&k=0,1.
\end{aligned}
\end{equation}
\end{lemma}

其中，$\gamma_{i,y}^{(k)}$ 是从 $x_i^{(k)}[S]$ 运到 $y[S]$ 的质量，
$p_k(y)$ 为运输后相应目标点的质量。保留全部目标点，包括源观测自身。
若投影后的目标点重合，则合并其质量。
目标函数的凹性和正齐次性保证合并不会使其减小；反过来，对两类采用相同比例拆分合并后的质量，
不会改变目标值。因此，固定的目标标签 $y\in\mathcal X$ 不影响该值。

以 $g_y$ 表示赋给 $y[S]$ 的评分。代入目标质量，得到
\[
\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\{p_0(y)\ell(-v)+p_1(y)\ell(v)\}
=\inf_g\left\{
\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\sum_{j,y}\gamma_{j,y}^{(1)}\ell(g_y)\right\}.
\]
运输可行集为紧凸集，花括号内的表达式关于有限维评分向量 $g$ 为凸函数，关于 $\gamma$ 为线性函数。
由有限维极小极大定理，可将有限维最不利分布问题的最优值表示为
\[
V(S)=\inf_g\left\{
\max_{\gamma^{(0)}}\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\max_{\gamma^{(1)}}\sum_{j,y}\gamma_{j,y}^{(1)}\ell(g_y)
\right\},
\]
其中，各最大化问题满足式~\eqref{eq:lfd} 中相应的源质量和运输预算约束。

固定 $g$ 时，第 $0$ 类的最大化问题是运输线性规划。
为源质量守恒约束分配自由乘子 $a_i^{(0)}$，为运输预算分配非负乘子 $\lambda_0$。
其拉格朗日函数为
\begin{equation}\label{eq:transport-lagrangian}
\begin{aligned}
\mathcal L_0={}&\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\sum_i a_i^{(0)}\left(\frac1{n_0}-\sum_y\gamma_{i,y}^{(0)}\right)\\
&+\lambda_0\left(\rho_0-\sum_{i,y}\gamma_{i,y}^{(0)}
\|x_i^{(0)}[S]-y[S]\|_1\right)\\
={}&\frac1{n_0}\sum_i a_i^{(0)}+\rho_0\lambda_0\\
&+\sum_{i,y}\gamma_{i,y}^{(0)}
\left[\ell(-g_y)-a_i^{(0)}
-\lambda_0\|x_i^{(0)}[S]-y[S]\|_1\right].
\end{aligned}
\end{equation}
对 $\gamma^{(0)}\ge0$ 取上确界时，其值有限当且仅当每个方括号中的系数均非正。
对两类分别应用线性规划对偶性，再对 $g$ 最小化，得到
\begin{equation}\label{eq:finite}
\begin{aligned}
V(S)=\min_{g,a,\lambda}\quad&
\frac1{n_0}\sum_i a_i^{(0)}+\frac1{n_1}\sum_j a_j^{(1)}
+\rho_0\lambda_0+\rho_1\lambda_1\\
\text{约束为}\quad&
\ell(-g_y)\le a_i^{(0)}+\lambda_0\|x_i^{(0)}[S]-y[S]\|_1,
&&i\in[n_0],\ y\in\mathcal X,\\
&\ell(g_y)\le a_j^{(1)}+\lambda_1\|x_j^{(1)}[S]-y[S]\|_1,
&&j\in[n_1],\ y\in\mathcal X,\\
&g\in\mathbb R^{\mathcal X},\quad a^{(k)}\in\mathbb R^{n_k},
\quad\lambda_k\ge0,&&k=0,1.
\end{aligned}
\end{equation}
将该目标函数记为 $J(a,\lambda)$。投影重合的目标点对应相同的约束列，
因此可以令其评分相等。附录~\ref{app:recovery} 将这些有限个评分延拓为
第~\ref{sec:problem} 节中的可测检测函数 $g$，并证明最小值可以取得。

零检测函数的风险为 $2$。在约束中取 $y=x_i^{(k)}$，得到 $a_i^{(k)}\ge0$，
因此最优解满足
\begin{equation}\label{eq:uniform-bound}
\rho_0\lambda_0+\rho_1\lambda_1\le J(a,\lambda)=V(S)\le2.
\end{equation}
利用这一上界可以引入坐标选择，并保持二元点处的最优值不变。

\section{选择变量与对偶}\label{sec:selection}
令 $z=\mathbf1_S$，并记
$\mathcal Z_d=\{z\in\{0,1\}^D:\mathbf1^\top z\le d\}$。
运输成本可以直接按坐标分解：
\[
\|x_i^{(k)}[S]-y[S]\|_1
=\sum_{m=1}^D z_m|x_i^{(k)}[m]-y[m]|.
\]
引入 $w_{km}$ 以替代 $\lambda_kz_m$。条件
$0\le w_{km}\le\lambda_k$ 和 $\rho_0w_{0m}+\rho_1w_{1m}\le2z_m$
使未选坐标的两类权重均为零。对于已选坐标，由式~\eqref{eq:uniform-bound}，
在最优解处可以取 $w_{km}=\lambda_k$。对于 $z\in[0,1]^D$，定义
\begin{equation}\label{eq:value}
\begin{aligned}
F(z)=\min_{g,a,\lambda,w}\quad&J(a,\lambda)\\
\text{约束为}\quad&
\ell(-g_y)\le a_i^{(0)}+\sum_m w_{0m}|x_i^{(0)}[m]-y[m]|,
&&i\in[n_0],\ y\in\mathcal X,\\
&\ell(g_y)\le a_j^{(1)}+\sum_m w_{1m}|x_j^{(1)}[m]-y[m]|,
&&j\in[n_1],\ y\in\mathcal X,\\
&0\le w_{km}\le\lambda_k,&&k=0,1,\ m\in[D],\\
&\rho_0w_{0m}+\rho_1w_{1m}\le2z_m,&&m\in[D],\\
&g\in\mathbb R^{\mathcal X},\quad a^{(k)}\in\mathbb R^{n_k},
\quad\lambda_k\ge0,&&k=0,1.
\end{aligned}
\end{equation}
该模型具有文献 \cite{bertsimas2021unified} 中问题~(2.1)--(2.2) 的逻辑结构：
外层问题选择坐标，每个坐标激活凸内层问题中的一个变量对 $(w_{0m},w_{1m})$。

\begin{proposition}[选择变量表示]\label{prop:selection}
式~\eqref{eq:value} 的最小值可以取得。函数 $F$ 在 $[0,1]^D$ 上为凸函数，且
\begin{equation}\label{eq:binary-exact}
F(\mathbf1_S)=V(S),\qquad
\min_{|S|\le d}V(S)=\min_{z\in\mathcal Z_d}F(z).
\end{equation}
\end{proposition}
\begin{proof}
式~\eqref{eq:finite} 的最优解可以扩展为式~\eqref{eq:value} 的可行解：
在 $z=\mathbf1_S$ 处令 $w_{km}=\lambda_kz_m$，可行性由式~\eqref{eq:uniform-bound} 保证。
反过来，该二元选集处的可行点在 $S$ 之外满足 $w_{km}=0$，故
\[
\sum_m w_{km}|x_i^{(k)}[m]-y[m]|
\le\lambda_k\sum_{m\in S}|x_i^{(k)}[m]-y[m]|
=\lambda_k\|x_i^{(k)}[S]-y[S]\|_1.
\]
因此，该点对式~\eqref{eq:finite} 可行，目标值不变，从而二元点处的最优值相等。
对于任意 $z$，可行解 $g=0$、$a=\mathbf1$、$\lambda=w=0$ 的目标值为 $2$。
由附录~\ref{app:recovery} 中的界及 $0\le w_{km}\le\lambda_k$，子水平集 $J\le2$ 紧致。
所有约束关于包括 $z$ 在内的变量联合凸。对可行解取凸组合，得到
\[
F(tz+(1-t)z')\le tF(z)+(1-t)F(z'),\qquad 0\le t\le1.
\]
\end{proof}

\subsection{对偶表示}\label{sec:dual}
为式~\eqref{eq:value} 中两组损失约束分别分配乘子
$\gamma_{i,y}^{(0)}\ge0$ 和 $\gamma_{j,y}^{(1)}\ge0$，为每个坐标容量约束分配乘子 $\mu_m\ge0$。
对自由源变量 $a$ 最小化，恢复源质量等式 $\sum_y\gamma_{i,y}^{(k)}=1/n_k$。
因此，同一运输记号通过对偶性重新出现。沿用式~\eqref{eq:lfd} 中的
$p_k(y)=\sum_{i=1}^{n_k}\gamma_{i,y}^{(k)}$，令
\begin{equation}\label{eq:dual-loss}
H(\gamma)=\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\{p_0(y)\ell(-v)+p_1(y)\ell(v)\}.
\end{equation}
这正是最不利分布问题的目标，由对目标点评分最小化得到。
其余对偶约束定义集合
\begin{equation}\label{eq:dual-domain}
\mathcal D=\left\{(\gamma,\mu)\ge0:
\begin{aligned}
&\sum_{y\in\mathcal X}\gamma_{i,y}^{(k)}=1/n_k,
&&i\in[n_k],\ k=0,1,\\
&\sum_{m=1}^D\left(
\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|-\rho_k\mu_m
\right)_+\le\rho_k,
&&k=0,1
\end{aligned}\right\},
\end{equation}
其中 $(v)_+=\max\{v,0\}$。附录~\ref{app:dual} 通过对 $\lambda,w$ 最小化推导第二行。
特别地，$\mathcal D$ 不依赖于 $z$。

\begin{proposition}[对偶表示]\label{prop:dual}
对于每个 $z\in[0,1]^D$，以下最大值可以取得：
\begin{equation}\label{eq:dual}
F(z)=\max_{(\gamma,\mu)\in\mathcal D}
\{H(\gamma)-2\mu^\top z\}.
\end{equation}
$z^t$ 处的最优解 $(\gamma^t,\mu^t)$ 给出支撑函数
\begin{equation}\label{eq:cut}
c_t(z)=H(\gamma^t)-2(\mu^t)^\top z
=F(z^t)-2(\mu^t)^\top(z-z^t)\le F(z),
\end{equation}
并满足 $c_t(z^t)=F(z^t)$。
\end{proposition}

函数 $H$ 是线性函数族的下确界，因而为凹函数，而 $\mathcal D$ 为凸集。
因此，固定 $z$ 时，对偶为凸优化问题。它关于 $z$ 的仿射形式给出了
文献 \cite{bertsimas2021unified} 中定理~2.5 和算法~3.1 所依据的切平面构造。

\section{切平面算法}\label{sec:algorithm}
记 $\mathrm{OPT}=\min_{z\in\mathcal Z_d}F(z)$。
评估 $z^1,\ldots,z^t$ 并取得相应的对偶最优解后，
令 $U_t=\min_{s\le t}F(z^s)$，并求解
\begin{equation}\label{eq:master}
\begin{aligned}
(z^{t+1},L_t)\in\operatorname*{arg\,min}_{z\in\mathcal Z_d,\,\eta\ge0}
\quad&\eta\\
\text{约束为}\quad&
\eta\ge F(z^s)-2(\mu^s)^\top(z-z^s),\qquad s=1,\ldots,t.
\end{aligned}
\end{equation}
每条保留的割均为全局下界，并在生成该割的已评估选集处取等号。因此，
\begin{equation}\label{eq:bounds}
L_t\le\mathrm{OPT}\le U_t.
\end{equation}
从任意 $z^1\in\mathcal Z_d$ 出发，交替求解凸子问题
\eqref{eq:value}--\eqref{eq:dual} 与二元线性主问题~\eqref{eq:master}。
当 $U_t-L_t\le\varepsilon$ 时停止，并返回最好的已评估选集。
附录~\ref{app:recovery} 恢复其检测函数，风险为 $U_t$，与最优风险的差距至多为 $\varepsilon$。
若内层求解非精确，使用实际可行对偶表达式 $H(\gamma^t)-2(\mu^t)^\top z$ 作为割，
以原始可行解的目标值作为上界；主问题必须提供全局下界。
该表示在二元点处精确，但其连续松弛可能较弱。

\section{有限维重构的证明}\label{app:recovery}
对于式~\eqref{eq:finite} 的任意可行解 $(g,a,\lambda)$，令投影重合的目标点评分相等。
这保持可行性与目标值不变。沿用文献 \cite[定理~1]{xie2021robust} 在样本支撑之外的延拓思路，
我们在每个位置同时控制两类损失。对于当前凸生成损失，构造满足以下条件的延拓：
\begin{equation}\label{eq:envelopes}
\begin{aligned}
\ell(-g(\xi))&\le E_0(\xi),&
\ell(g(\xi))&\le E_1(\xi),\\
E_k(\xi)&=\min_{i\in[n_k]}
\{a_i^{(k)}+\lambda_k\|x_i^{(k)}[S]-\xi\|_1\},
&&k=0,1.
\end{aligned}
\end{equation}
\begin{proof}[引理~\ref{prop:lfd} 的证明]
在观测投影处令 $g(y[S])=g_y$；式~\eqref{eq:finite} 的约束保证式~\eqref{eq:envelopes} 在此处成立。
在其他位置，分别选择达到 $E_0(\xi),E_1(\xi)$ 的最小下标 $i,j$，并令
\[
r_0=\|x_i^{(0)}[S]-\xi\|_1,\qquad
r_1=\|x_j^{(1)}[S]-\xi\|_1,\qquad
b=\|x_i^{(0)}[S]-x_j^{(1)}[S]\|_1.
\]
当 $b>0$ 时，令 $\tau=\min\{1,r_0/b\}$，并定义
\begin{equation}\label{eq:extension}
g(\xi)=(1-\tau)g_{x_i^{(0)}}+\tau g_{x_j^{(1)}}.
\end{equation}
三角不等式给出 $\tau b\le r_0$ 和 $(1-\tau)b\le r_1$。
由凸性及式~\eqref{eq:finite}，得到
\[
\begin{aligned}
\ell(-g(\xi))
&\le(1-\tau)\ell(-g_{x_i^{(0)}})+\tau\ell(-g_{x_j^{(1)}})
\le a_i^{(0)}+\lambda_0\tau b\le E_0(\xi),\\
\ell(g(\xi))
&\le(1-\tau)\ell(g_{x_i^{(0)}})+\tau\ell(g_{x_j^{(1)}})
\le a_j^{(1)}+\lambda_1(1-\tau)b\le E_1(\xi).
\end{aligned}
\]
当 $b=0$ 时，取 $g(\xi)=g_{x_i^{(0)}}$，由式~\eqref{eq:finite} 得到相同上界。
有限个下标中的确定性择取规则保证 $g$ 为 Borel 可测函数，且 $|g|\le\max_{y\in\mathcal X}|g_y|$。
对于任意 $\mathbb P_k\in\mathcal P_k$，对从经验分布到 $\mathbb P_k$ 的最优耦合积分式~\eqref{eq:envelopes}，
并利用投影不增加成本，得到
\[
\begin{aligned}
\mathbb E_{\mathbb P_0}\ell(-g(X[S]))
&\le\frac1{n_0}\sum_i a_i^{(0)}+\rho_0\lambda_0,\\
\mathbb E_{\mathbb P_1}\ell(g(X[S]))
&\le\frac1{n_1}\sum_j a_j^{(1)}+\rho_1\lambda_1.
\end{aligned}
\]
因此，$f(x)=g(x[S])$ 满足 $R(f)\le J(a,\lambda)$。
在非空子水平集 $J\le2$ 上，有 $0\le a_i^{(k)}\le2n_k$ 和 $0\le\lambda_k\le2/\rho_k$。
每个 $g_y$ 的两类损失均由相应类别的约束控制。
由于当 $t\to\infty$ 时有 $\ell(t)\to\infty$，所有评分均有界。
该子水平集紧致，故式~\eqref{eq:finite} 取得最小值。
由弱极小极大不等式，并将运输支撑限制在有限观测点上，
式~\eqref{eq:lfd} 的有限维最不利分布最优值给出 $V(S)$ 的下界。
有限维极小极大与运输线性规划对偶表明，该值等于式~\eqref{eq:finite} 的最小值。
上述延拓给出反向不等式 $V(S)\le\min J(a,\lambda)$。
因此，两种有限维表示均等于 $V(S)$，且最优解的延拓达到这一风险。
\end{proof}

\section{对偶表示的证明}\label{app:dual}
\begin{proof}[命题~\ref{prop:dual} 的证明]
除 $\gamma,\mu$ 外，为 $w_{km}\le\lambda_k$ 分配乘子 $\beta_{km}\ge0$。
保留 $g,a$ 为自由变量、$\lambda,w\ge0$，最小化问题的拉格朗日函数为
\begin{equation}\label{eq:full-lagrangian}
\begin{aligned}
\mathcal L={}&J(a,\lambda)
+\sum_{i,y}\gamma_{i,y}^{(0)}
\left[\ell(-g_y)-a_i^{(0)}
-\sum_m w_{0m}|x_i^{(0)}[m]-y[m]|\right]\\
&+\sum_{j,y}\gamma_{j,y}^{(1)}
\left[\ell(g_y)-a_j^{(1)}
-\sum_m w_{1m}|x_j^{(1)}[m]-y[m]|\right]\\
&+\sum_{k,m}\beta_{km}(w_{km}-\lambda_k)
+\sum_m\mu_m(\rho_0w_{0m}+\rho_1w_{1m}-2z_m).
\end{aligned}
\end{equation}
对 $a$ 取下确界时，其值有限当且仅当 $\sum_y\gamma_{i,y}^{(k)}=1/n_k$。
与 $g$ 有关的项为 $\sum_y[p_0(y)\ell(-g_y)+p_1(y)\ell(g_y)]$，其下确界为 $H(\gamma)$。
$\lambda,w\ge0$ 的系数必须非负：
\[
\rho_k-\sum_m\beta_{km}\ge0,\qquad
\beta_{km}+\rho_k\mu_m
-\sum_{i,y}\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|\ge0.
\]
与 $\beta\ge0$ 合并后，这些条件等价于
\[
\sum_m\left(
\sum_{i,y}\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|
-\rho_k\mu_m\right)_+\le\rho_k.
\]
因此，消去 $\beta$ 后得到可行域 $\mathcal D$ 和目标函数 $H(\gamma)-2\mu^\top z$。

对于每个固定的 $z\in[0,1]^D$，点 $g=0$、$a=2\mathbf1$、$\lambda=\mathbf1$、$w=0$
严格满足所有损失不等式，并满足全部仿射约束。
将非负性视为仿射约束后，允许仿射约束取等号的 Slater 条件给出强对偶及对偶最优值的可达性，
也涵盖边界选集。最后，任意对偶可行点均给出全局下界。
$(\gamma^t,\mu^t)$ 在 $z^t$ 处的最优性给出
\[
F(z)\ge H(\gamma^t)-2(\mu^t)^\top z
=F(z^t)-2(\mu^t)^\top(z-z^t),
\]
这就证明了式~\eqref{eq:cut}。
\end{proof}
