\begin{abstract}
Wasserstein 鲁棒检验的变量选择问题具有精确有限表示及最大仿射对偶表示。
对偶耦合生成全局割。理论分析包括非精确求解的风险保证，以及依赖数据的选点数量界。
\end{abstract}

\section{问题设定}\label{sec:problem}
令 $[D]=\{1,\ldots,D\}$，$0\le d\le D$ 为整数。
对于可测评分函数 $f:\mathbb R^D\to\mathbb R$，定义
\begin{equation}\label{eq:risk}
R(f)=\sup_{\substack{P_0\in\mathcal U_0\\P_1\in\mathcal U_1}}
\left\{\mathbb E_{P_0}\ell(-f(X))+
       \mathbb E_{P_1}\ell(f(X))\right\}.
\end{equation}
损失 $\ell$ 有限、非负、凸且非减，并满足 $\ell(0)>0$。
记 $\ell_0(s)=\ell(-s)$，$\ell_1(s)=\ell(s)$。
对于 $S\subseteq[D]$，以 $\operatorname{Proj}_S:\mathbb R^D\to\mathbb R^{|S|}$
表示坐标投影，并定义
\[
\mathcal G_S=\{g:\mathbb R^{|S|}\to\mathbb R:\ g\text{ 为Borel可测}\}.
\]
变量选择问题为
\begin{equation}\label{eq:problem}
\boxed{V(S)=\inf_{g\in\mathcal G_S}R(g\circ\operatorname{Proj}_S),
\qquad \mathrm{OPT}=\min_{|S|\le d}V(S).}
\end{equation}

$N$ 个观测为 $x_1,\ldots,x_N\in\mathbb R^D$。
非空索引集 $I_0,I_1$ 按类别划分 $[N]$，类内概率质量为 $q_i>0$：
\[
\sum_{i\in I_k}q_i=1,\qquad
\widehat P_k=\sum_{i\in I_k}q_i\delta_{x_i},\qquad
\mathcal U_k=\{P:W_1(P,\widehat P_k)\le\theta_k\},\quad k=0,1.
\]
其中 $\delta_x$ 为点 $x$ 处的单位质量，$\theta_k>0$，
$W_1$ 对具有有限一阶矩的分布使用 $\ell_1$ 代价。
不另加权时，$i\in I_k$ 的样本取 $q_i=1/|I_k|$。

\section{对偶重构}\label{sec:value}
固定 $S$，记
\[
x_i^S=\operatorname{Proj}_Sx_i,\qquad \Delta_{ir,j}=|x_{ij}-x_{rj}|,\qquad
c_{ir}^S=\|x_i^S-x_r^S\|_1=\sum_{j\in S}\Delta_{ir,j}.
\]
其中 $i,r\in[N]$，$j\in[D]$。
耦合 $\pi_{ir}\ge0$ 将质量从源样本 $i$ 运输到目标样本 $r$。定义
\[
\mathcal T_k(S)=\left\{\pi\ge0:
\sum_r\pi_{ir}=q_i,\quad
\sum_{i\in I_k,r}\pi_{ir}c_{ir}^S\le\theta_k\right\}.
\]
以 $u=(u_1,\ldots,u_N)$ 为目标点评分变量，有限运输表示为
\begin{equation}\label{eq:finite-transport}
V(S)=\inf_{u\in\mathbb R^N}
\sum_{k=0}^1\max_{\pi\in\mathcal T_k(S)}
\sum_{i\in I_k,r}\pi_{ir}\ell_k(u_r).
\end{equation}
对各运输线性规划取对偶，引入源样本损失上界 $a_i\in\mathbb R$
和运输价格 $\lambda_k\ge0$。
记 $J(a,\lambda)=\sum_iq_i a_i+\sum_k\theta_k\lambda_k$，得到
\begin{equation}\label{eq:finite}
\begin{aligned}
V(S)=\min_{a,u\in\mathbb R^N,\ \lambda\ge0}\quad&J(a,\lambda)\\
\text{要求}\quad&
\ell_k(u_r)\le a_i+\lambda_kc_{ir}^S,
\quad i\in I_k,\ r\in[N],\ k=0,1.
\end{aligned}
\end{equation}
附录~\ref{app:finite} 证明精确性，并恢复最优检验。

对于二元指示向量 $z=\mathbf1_S$，用坐标价格 $w_{kj}$ 代替按选择开关的价格 $\lambda_kz_j$。
由于 $a_i\ge0$，
\[
\sum_k\theta_kw_k=z\sum_k\theta_k\lambda_k\le zJ(a,\lambda)
\quad\text{当 }w_k=\lambda_kz.
\]
固定 $z\in[0,1]^D$，定义
\begin{equation}\label{eq:value}
\begin{aligned}
F(z)=\min_{a,u,\lambda,w}\quad&J(a,\lambda)\\
\text{要求}\quad&
\ell_k(u_r)\le a_i+\sum_j\Delta_{ir,j}w_{kj},
\quad i\in I_k,\ r\in[N],\ k=0,1,\\
&0\le w_k\le\lambda_k\mathbf1,\qquad k=0,1,\\
&\theta_0w_0+\theta_1w_1\le zJ(a,\lambda),\\
&\lambda\ge0,\qquad a,u\in\mathbb R^N.
\end{aligned}
\end{equation}
其中 $w_k=(w_{k1},\ldots,w_{kD})^\top$，$\mathbf1=(1,\ldots,1)^\top$；
向量不等式按坐标理解。对于二元 $z$，在其余约束下，
\[
\theta_0w_0+\theta_1w_1\le zJ
\quad\Longleftrightarrow\quad
w_{kj}=0\quad\text{当 }z_j=0,\ k=0,1.
\]
当 $z_j=1$ 时，该坐标上的这条不等式冗余，可以省略：
\[
\theta_0w_{0j}+\theta_1w_{1j}
\le\sum_k\theta_k\lambda_k\le J.
\]
删除整组约束则也会移除未选坐标上的限制，使每个 $z$ 都对应 $V([D])$。
保留它则有 $F(\mathbf1_S)=V(S)$；命题~\ref{prop:binary} 证明投影后的可行域相同，且目标保持不变。

式~\eqref{eq:value} 的对偶使用经缩放的运输耦合 $\Pi\ge0$、
标量 $\alpha\ge0$ 及坐标乘子 $\mu\ge0$，其形式为
\begin{equation}\label{eq:dual-normalized}
\begin{aligned}
F(z)=\max_{\Pi,\alpha,\mu}\quad&H(\Pi)\\
\text{要求}\quad&(\Pi,\alpha,\mu)\in\mathcal D,\qquad
\alpha+\mu^\top z=1.
\end{aligned}
\end{equation}
其中
\[
H(\Pi)=\sum_{r=1}^N\inf_{s\in\mathbb R}
\sum_{k=0}^1\sum_{i\in I_k}\Pi_{ir}\ell_k(s),
\]
并以 $(v)_+=\max\{v,0\}$ 记正部，有
\begin{equation}\label{eq:dual-domain}
\mathcal D=\left\{(\Pi,\alpha,\mu)\ge0:
\begin{array}{l}
\displaystyle\sum_r\Pi_{ir}=\alpha q_i,\quad i\in[N],\\[3pt]
\displaystyle\sum_j
\left(\sum_{i\in I_k,r}\Pi_{ir}\Delta_{ir,j}-\theta_k\mu_j\right)_+
\le\alpha\theta_k,\quad k=0,1
\end{array}\right\}.
\end{equation}
该耦合的源质量为 $\alpha q_i$；$H$ 为每个目标点选取一个评分，
使其加权损失最小。拉格朗日推导见命题~\ref{prop:normalized}。

对于归一化三元组 $\alpha+\mu^\top z=1$，
将三个对偶变量同时乘以 $\tau\ge0$ 仍属于 $\mathcal D$，并有
\[
H(\tau\Pi)=\tau H(\Pi),\qquad
\max_{\tau\ge0}\{2\sqrt{\tau H(\Pi)}-\tau\}=H(\Pi),
\quad \tau^*=H(\Pi).
\]
在每条射线上优化尺度，可消去归一化约束：
\begin{equation}\label{eq:dual-affine}
\boxed{F(z)=\max_{(\Pi,\alpha,\mu)\in\mathcal D}
\{2\sqrt{H(\Pi)}-\alpha-\mu^\top z\}.}
\end{equation}
命题~\ref{prop:dual} 证明这一等价性及最优解的存在性。归一化最优解 $(\bar\Pi,\bar\alpha,\bar\mu)$ 对应的仿射最优解为 $F(z)(\bar\Pi,\bar\alpha,\bar\mu)$；割的斜率也必须乘以这一尺度。

\section{切平面算法}\label{sec:algorithm}
令 $\mathcal Z_d=\{z\in\{0,1\}^D:\mathbf1^\top z\le d\}$。
在 $z^t$ 处用式~\eqref{eq:dual-affine} 评估 $F(z^t)$，
取满足命题~\ref{prop:bounded} 中界 $\mu_j^t\le M_j$ 的最优解
$(\Pi^t,\alpha^t,\mu^t)$。其次梯度 $-\mu^t$ 给出
\begin{equation}\label{eq:tangent}
F(z)\ge2\sqrt{H(\Pi^t)}-\alpha^t-(\mu^t)^\top z
=F(z^t)-(\mu^t)^\top(z-z^t).
\end{equation}
下一选集和下界由以下问题给出
\begin{equation}\label{eq:master}
\begin{aligned}
(z^{t+1},\eta^{t+1})\in
\operatorname*{arg\,min}_{z\in\mathcal Z_d,\ \eta\ge0}\quad&\eta\\
\text{要求}\quad&
\eta\ge F(z^s)-(\mu^s)^\top(z-z^s),
\qquad s=1,\ldots,t.
\end{aligned}
\end{equation}
在精确求解下，记 $U_t=\min_{s\le t}F(z^s)$，$T$ 为最后一次迭代，
附录~\ref{app:convergence} 证明
\begin{equation}\label{eq:certificate}
\begin{aligned}
&\eta^{t+1}\le\mathrm{OPT}\le U_t,\\
&0\le R(f^*)-\mathrm{OPT}\le U_T-\eta^{T+1}\le\varepsilon,
\qquad T\le\sum_{r=0}^d\binom Dr.
\end{aligned}
\end{equation}

非精确求解时，可行评分提供上界 $U_t$，对偶可行割仍是全局下界，
主问题的全局下界 $L_t$ 给出
\[
0\le R(f_{\rm inc})-\mathrm{OPT}\le U_t-L_t.
\]
命题~\ref{prop:inexact} 证明这一保证，以及有限终止的充分精度条件
$\delta+\beta\le\varepsilon$。
命题~\ref{prop:bounded} 和 \ref{prop:convergence} 进一步给出有界斜率，
以及精确搜索中依赖数据的选点间隔界。

\begin{algorithm}[htbp]
\caption{变量选择的精确切平面算法}\label{alg:oa}
\begin{algorithmic}[1]
\Require 问题数据、预算 $d$、容差 $\varepsilon>0$ 和 $z^1\in\mathcal Z_d$。
\Ensure 选集 $S^*$、其最优检验 $f^*$，以及宽度至多为 $\varepsilon$ 的风险区间。
\State $t\gets1$，$U\gets+\infty$。
\While{真}
  \State 在 $z^t$ 处精确求解式~\eqref{eq:dual-affine}；按命题~\ref{prop:bounded} 选取满足 $0\le\mu_j^t\le M_j$ 的 $\mu^t$。
  \If{$F(z^t)<U$}
    \State $(U,z^*)\gets(F(z^t),z^t)$。
  \EndIf
  \State 保留索引为 $t$ 的割~\eqref{eq:tangent}。
  \State 精确求解式~\eqref{eq:master}，得到 $(z^{t+1},\eta^{t+1})$。
  \If{$U-\eta^{t+1}\le\varepsilon$}
    \State $S^*\gets\{j:z_j^*=1\}$。
    \State 对 $S^*$ 求解式~\eqref{eq:finite}，按式~\eqref{eq:extension} 构造 $g^*$，并令 $f^*=g^*\circ\operatorname{Proj}_{S^*}$。
    \State \Return $(S^*,f^*,[\eta^{t+1},U])$。
  \EndIf
  \State $t\gets t+1$。
\EndWhile
\end{algorithmic}
\end{algorithm}

\section{有限表示的证明}\label{app:finite}
设 $x_i\in\mathbb R^D$，$i\in[N]$，非空集合 $I_0,I_1$ 划分 $[N]$。
设 $q_i>0$，$\sum_{i\in I_k}q_i=1$，$\theta_k>0$。
损失 $\ell:\mathbb R\to[0,\infty)$ 有限、凸且非减，满足 $\ell(0)>0$；
记 $\ell_0(s)=\ell(-s)$，$\ell_1(s)=\ell(s)$。
对于 $S\subseteq[D]$，以 $\operatorname{Proj}_S$ 表示坐标投影，
以 $\delta_x$ 表示点 $x$ 处的单位质量。定义
\[
\begin{gathered}
x_i^S=\operatorname{Proj}_Sx_i,\qquad
c_{ir}^S=\|x_i^S-x_r^S\|_1,\qquad
\widehat P_k=\sum_{i\in I_k}q_i\delta_{x_i},\\
\mathcal U_k=\{P:W_1(P,\widehat P_k)\le\theta_k\},\qquad
\mathcal G_S=\{g:\mathbb R^{|S|}\to\mathbb R:\ g\text{ 为Borel可测}\},\\
R(f)=\sum_{k=0}^1\sup_{P\in\mathcal U_k}\mathbb E_P\ell_k(f(X)),
\qquad
V(S)=\inf_{g\in\mathcal G_S}R(g\circ\operatorname{Proj}_S).
\end{gathered}
\]
其中 $W_1$ 对有限一阶矩分布使用 $\ell_1$ 运输代价。设
\begin{equation}\label{eq:appendix-finite}
\begin{aligned}
T(S)=\inf_{a,u\in\mathbb R^N,\ \lambda\in\mathbb R_+^2}\quad&
J(a,\lambda):=\sum_iq_i a_i+\sum_{k=0}^1\theta_k\lambda_k\\
\text{要求}\quad&
\ell_k(u_r)\le a_i+\lambda_kc_{ir}^S,
\quad i\in I_k,\ r\in[N],\ k=0,1.
\end{aligned}
\end{equation}
在证明与 $V(S)$ 相等之前，以 $T(S)$ 单独表示有限问题的值。

\begin{proposition}\label{prop:finite}
对于每个 $S\subseteq[D]$，有 $V(S)=T(S)$，定义 $T(S)$ 的下确界可以取到。
每个有限最优解都可构造一个 Borel 评分函数
$g^*:\mathbb R^{|S|}\to\mathbb R$，满足
$R(g^*\circ\operatorname{Proj}_S)=V(S)$。
\end{proposition}
\begin{proof}
\textit{有限运输对偶。}
固定 $u\in\mathbb R^N$，定义
\[
\mathcal T_k(S)=\left\{\pi\in\mathbb R_+^{|I_k|\times N}:
\sum_r\pi_{ir}=q_i\ (i\in I_k),\quad
\sum_{i\in I_k,r}\pi_{ir}c_{ir}^S\le\theta_k\right\}.
\]
对角计划可行，且 $0\le\pi_{ir}\le q_i$，故 $\mathcal T_k(S)$ 非空紧致。
取 $a_i\in\mathbb R$、$\lambda_k\ge0$，以下 $i\in I_k$、$r\in[N]$：
\begin{equation}\label{eq:transport-lagrangian}
\begin{aligned}
L_k(\pi,a,\lambda_k;u)
={}&\sum_{i,r}\pi_{ir}\ell_k(u_r)
+\sum_i a_i\left(q_i-\sum_r\pi_{ir}\right)
+\lambda_k\left(\theta_k-\sum_{i,r}\pi_{ir}c_{ir}^S\right)\\
={}&\sum_iq_i a_i+\theta_k\lambda_k
+\sum_{i,r}\pi_{ir}[\ell_k(u_r)-a_i-\lambda_kc_{ir}^S].
\end{aligned}
\end{equation}
\[
\sup_{\pi\ge0}L_k=
\begin{cases}
\sum_iq_i a_i+\theta_k\lambda_k,
&\ell_k(u_r)\le a_i+\lambda_kc_{ir}^S\quad\forall i,r,\\
+\infty,&\text{其他情形}.
\end{cases}
\]
由线性规划强对偶，
\begin{equation}\label{eq:transport-dual}
\begin{aligned}
\max_{\pi\in\mathcal T_k(S)}\sum_{i\in I_k,r}\pi_{ir}\ell_k(u_r)
=\min_{(a_i)_{i\in I_k},\ \lambda_k\ge0}\quad&
\sum_{i\in I_k}q_i a_i+\theta_k\lambda_k\\
\text{要求}\quad&
\ell_k(u_r)\le a_i+\lambda_kc_{ir}^S,\quad i\in I_k,\ r\in[N].
\end{aligned}
\end{equation}
\[
T(S)=\inf_{u\in\mathbb R^N}\sum_{k=0}^1
\max_{\pi\in\mathcal T_k(S)}\sum_{i\in I_k,r}\pi_{ir}\ell_k(u_r).
\]
这里只对有限线性规划取对偶，未交换极小与极大的顺序。

\textit{下界：$T(S)\le V(S)$。}
对 $g\in\mathcal G_S$、$\pi\in\mathcal T_k(S)$，取
$\widetilde x_{ir}\in\mathbb R^D$ 满足
\[
\operatorname{Proj}_S\widetilde x_{ir}=x_r^S,\qquad
\operatorname{Proj}_{S^c}\widetilde x_{ir}=\operatorname{Proj}_{S^c}x_i,
\qquad \|\widetilde x_{ir}-x_i\|_1=c_{ir}^S.
\]
\begin{equation}\label{eq:expectation-transport}
\begin{aligned}
P_k^\pi&=\sum_{i\in I_k,r}\pi_{ir}\delta_{\widetilde x_{ir}},
&W_1(P_k^\pi,\widehat P_k)&\le\sum_{i\in I_k,r}\pi_{ir}c_{ir}^S\le\theta_k,\\
\mathbb E_{P_k^\pi}\ell_k(g(\operatorname{Proj}_S X))
&=\sum_{i\in I_k,r}\pi_{ir}\ell_k(g(x_r^S)).
\end{aligned}
\end{equation}
\begin{equation}\label{eq:finite-lower}
\begin{aligned}
V(S)&\ge\inf_{g\in\mathcal G_S}\sum_k
\max_{\pi\in\mathcal T_k(S)}\sum_{i\in I_k,r}\pi_{ir}\ell_k(g(x_r^S))\\
&\ge\inf_{u\in\mathbb R^N}\sum_k
\max_{\pi\in\mathcal T_k(S)}\sum_{i\in I_k,r}\pi_{ir}\ell_k(u_r)=T(S).
\end{aligned}
\end{equation}
\textit{上界：$V(S)\le T(S)$。}
对任意有限可行解 $(a,u,\lambda)$ 与 $y\in\mathbb R^{|S|}$，令
\[
\begin{gathered}
E_k(y)=\min_{i\in I_k}\{a_i+\lambda_k\|x_i^S-y\|_1\},\qquad
i_k(y)=\min\operatorname*{arg\,min}_{i\in I_k}
\{a_i+\lambda_k\|x_i^S-y\|_1\},\\
r_k=\|x_{i_k}^S-y\|_1,\qquad
\rho=\|x_{i_0}^S-x_{i_1}^S\|_1\le r_0+r_1.
\end{gathered}
\]
\begin{equation}\label{eq:extension}
g(y)=\begin{cases}
(1-\gamma)u_{i_0}+\gamma u_{i_1},\quad
\gamma=\min\{1,r_0/\rho\},&\rho>0,\\
u_{i_0},&\rho=0.
\end{cases}
\end{equation}
当 $\rho>0$ 时，$\gamma\rho\le r_0$、$(1-\gamma)\rho\le r_1$。
由凸性与有限可行性，
\[
\begin{aligned}
\ell_0(g(y))&\le(1-\gamma)\ell_0(u_{i_0})+\gamma\ell_0(u_{i_1})
\le a_{i_0}+\lambda_0\gamma\rho\le E_0(y),\\
\ell_1(g(y))&\le(1-\gamma)\ell_1(u_{i_0})+\gamma\ell_1(u_{i_1})
\le a_{i_1}+\lambda_1(1-\gamma)\rho\le E_1(y).
\end{aligned}
\]
当 $\rho=0$ 时，源与目标对 $(i_0,i_0)$、$(i_1,i_0)$ 给出相同的两个界。
最小指标规则保证 $g$ 为 Borel 可测，且 $|g(y)|\le\max_r|u_r|$；无须插值所有 $u_r$。
对 $P\in\mathcal U_k$，将最优耦合分解为源测度 $\nu_i$（重复源按 $q_i$ 拆分）：
\[
\nu_i(\mathbb R^D)=q_i,\qquad
\sum_{i\in I_k}\nu_i=P,\qquad
\sum_{i\in I_k}\int\|x-x_i\|_1\,\nu_i(dx)\le\theta_k.
\]
\begin{equation}\label{eq:extension-risk}
\begin{aligned}
\mathbb E_P\ell_k(g(\operatorname{Proj}_S X))
&\le\sum_{i\in I_k}\int
[a_i+\lambda_k\|\operatorname{Proj}_Sx-x_i^S\|_1]\,\nu_i(dx)\\
&\le\sum_{i\in I_k}q_i a_i+\theta_k\lambda_k.
\end{aligned}
\end{equation}
\[
R(g\circ\operatorname{Proj}_S)\le J(a,\lambda),\qquad V(S)\le T(S).
\]
\textit{最优解存在性。}
可行点 $(a,u,\lambda)=(\ell(0)\mathbf1,0,0)$ 的目标为 $C=2\ell(0)$。
由对角可行性与 $J\le C$，
\[
0\le a_i\le C/q_i,\qquad 0\le\lambda_k\le C/\theta_k,\qquad
\ell_k(u_r)\le\min_{i\in I_k}\{C/q_i+(C/\theta_k)c_{ir}^S\}.
\]
若 $\ell$ 非常数，则凸性与非减性给出 $t\to\infty$ 时 $\ell(t)\to\infty$，
两类损失分别界定 $u$ 的两侧。常数损失时取 $u=0$。
因此该闭子水平集存在非空紧致限制，包含最优解；其延拓达到 $V(S)$。
\end{proof}

\section{三个对偶重构的证明}\label{app:dual}
设 $I_0,I_1$ 划分 $[N]$，$q_i>0$，$\sum_{i\in I_k}q_i=1$，
$\theta_k>0$。
设 $\ell:\mathbb R\to[0,\infty)$ 有限、凸且非减，满足 $\ell(0)>0$，
记 $\ell_0(s)=\ell(-s)$，$\ell_1(s)=\ell(s)$。
对于数据 $x_i\in\mathbb R^D$，设
\[
\Delta_{ir,j}=|x_{ij}-x_{rj}|,\qquad
c_{ir}^S=\sum_{j\in S}\Delta_{ir,j},\qquad
J(a,\lambda)=\sum_iq_i a_i+\sum_k\theta_k\lambda_k.
\]
定义有限问题的可行域
\[
\mathcal C_S=\{(a,u,\lambda):
a,u\in\mathbb R^N,\ \lambda\ge0,\
\ell_k(u_r)\le a_i+\lambda_kc_{ir}^S
\ (i\in I_k,\ r\in[N],\ k=0,1)\}.
\]
设 $V(S):=\min_{\mathcal C_S}J$。
由命题~\ref{prop:finite}，这一值等于原固定选集分布问题的风险。
对于 $z\in[0,1]^D$，定义
\begin{equation}\label{eq:appendix-extension}
\begin{aligned}
\mathcal E_z=\{(a,u,\lambda,w):\quad&
a,u\in\mathbb R^N,\ \lambda\ge0,\quad
0\le w_{kj}\le\lambda_k,\\
&\ell_k(u_r)\le a_i+\sum_j\Delta_{ir,j}w_{kj}
\ (i\in I_k,\ r\in[N],\ k=0,1),\\
&\sum_k\theta_kw_{kj}\le z_jJ(a,\lambda)\ (j\in[D])\},
\qquad F(z)=\min_{\mathcal E_z}J.
\end{aligned}
\end{equation}

\begin{proposition}[二元扩展的精确性]\label{prop:binary}
对于每个 $S\subseteq[D]$，将扩展可行域投影到原变量后，有
\[
\operatorname{Proj}_{(a,u,\lambda)}\mathcal E_{\mathbf1_S}
=\mathcal C_S,\qquad F(\mathbf1_S)=V(S).
\]
两个方向均保持同一个目标函数 $J(a,\lambda)$。
\end{proposition}
\begin{proof}
令 $z=\mathbf1_S$。对 $(a,u,\lambda)\in\mathcal C_S$，
对角可行性给出 $a_i\ge0$。取 $w_k=\lambda_kz$，有
\[
\ell_k(u_r)\le a_i+\sum_j\Delta_{ir,j}w_{kj},\qquad
\sum_k\theta_kw_{kj}=z_j\sum_k\theta_k\lambda_k\le z_jJ(a,\lambda).
\]
因此 $(a,u,\lambda,w)\in\mathcal E_z$，且 $J$ 不变。
反之，$(a,u,\lambda,w)\in\mathcal E_z$ 给出
\[
w_{kj}=0\quad(j\notin S),\qquad
0\le w_{kj}\le\lambda_k\quad(j\in S),\qquad
\ell_k(u_r)\le a_i+\sum_{j\in S}\Delta_{ir,j}w_{kj}
\le a_i+\lambda_kc_{ir}^S.
\]
故 $(a,u,\lambda)\in\mathcal C_S$，且 $J$ 不变，两向包含即得结论。
选择约束在 $z_j=1$ 时冗余，在 $z_j=0$ 时强制 $w_{kj}=0$；
删除全部选择约束会使每个 $z$ 的值均为 $V([D])$。
\end{proof}

对于 $\Pi\in\mathbb R_+^{N\times N}$，定义
\[
H(\Pi)=\sum_{r=1}^N\inf_{s\in\mathbb R}
\sum_{k=0}^1\sum_{i\in I_k}\Pi_{ir}\ell_k(s),\qquad
d_{kj}(\Pi)=\sum_{i\in I_k,r}\Pi_{ir}\Delta_{ir,j}.
\]
$\Pi$ 是经缩放的运输耦合，其总质量不一定等于概率耦合的总质量。
以 $(v)_+=\max\{v,0\}$ 记正部，设
\begin{equation}\label{eq:appendix-cone}
\mathcal D=\left\{(\Pi,\alpha,\mu)\ge0:
\sum_r\Pi_{ir}=\alpha q_i\ (i\in[N]),\quad
\sum_j(d_{kj}(\Pi)-\theta_k\mu_j)_+
\le\alpha\theta_k\ (k=0,1)\right\}.
\end{equation}

\begin{proposition}[归一化对偶]\label{prop:normalized}
对于每个 $z\in[0,1]^D$，扩展问题取得最小值，且
\begin{equation}\label{eq:appendix-normalized}
0<F(z)=
\max_{\substack{(\Pi,\alpha,\mu)\in\mathcal D\\
\alpha+\mu^\top z=1}}H(\Pi)\le2\ell(0).
\end{equation}
归一化对偶的最大值也可以取到。
\end{proposition}
\begin{proof}
\textit{拉格朗日函数。}
分别对损失、价格上界和选择约束引入非负乘子 $\Pi_{ir},\beta_{kj},\mu_j$，
保留 $\lambda,w\ge0$。
\begin{equation}\label{eq:lagrangian-raw}
\begin{aligned}
\mathcal L={}&J(a,\lambda)
+\sum_{k,i\in I_k,r}\Pi_{ir}
\left[\ell_k(u_r)-a_i-\sum_j\Delta_{ir,j}w_{kj}\right]\\
&+\sum_{k,j}\beta_{kj}(w_{kj}-\lambda_k)
+\sum_j\mu_j\left[\sum_k\theta_kw_{kj}-z_jJ(a,\lambda)\right].
\end{aligned}
\end{equation}
记 $\alpha=1-\mu^\top z$，
\begin{equation}\label{eq:lagrangian}
\begin{aligned}
\mathcal L={}&\sum_i a_i\left(\alpha q_i-\sum_r\Pi_{ir}\right)
+\sum_{k,i\in I_k,r}\Pi_{ir}\ell_k(u_r)\\
&+\sum_k\lambda_k\left(\alpha\theta_k-\sum_j\beta_{kj}\right)
+\sum_{k,j}w_{kj}[\beta_{kj}+\theta_k\mu_j-d_{kj}(\Pi)].
\end{aligned}
\end{equation}
对自由变量 $a$ 及非负变量 $\lambda,w$ 的下确界有限，当且仅当
\begin{equation}\label{eq:dual-conditions}
\sum_r\Pi_{ir}=\alpha q_i,\qquad
\sum_j\beta_{kj}\le\alpha\theta_k,\qquad
\beta_{kj}\ge d_{kj}(\Pi)-\theta_k\mu_j.
\end{equation}
行和等式给出 $\alpha\ge0$。精确消去 $\beta\ge0$ 得
\[
\min_{\beta_{kj}\ge(d_{kj}(\Pi)-\theta_k\mu_j)_+}\sum_j\beta_{kj}
=\sum_j(d_{kj}(\Pi)-\theta_k\mu_j)_+.
\]
对 $u$ 取下确界得到 $H(\Pi)$，即得归一化对偶。

\textit{强对偶。}
记 $J_0=\{j:z_j=0\}$，可行性强制 $w_{kj}=0$。
删去这些变量及选择约束。约化问题的严格可行点为
\[
u=0,\qquad a_i=\ell(0)+1,\qquad\lambda_k=1,\qquad
w_{kj}=z_j/2\quad(j\notin J_0),
\]
\[
\ell_k(0)<a_i+\sum_j\Delta_{ir,j}w_{kj},\qquad
0<w_{kj}<\lambda_k,\qquad
\sum_k\theta_kw_{kj}<z_jJ(a,\lambda).
\]
当 $J_0=[D]$ 时省去 $w$，同样成立。
Slater 定理保证约化问题的强对偶及对偶最优解存在性。
对删除的坐标取
\[
\beta_{kj}=0,\qquad \mu_j=\max_{k=0,1}d_{kj}(\Pi)/\theta_k
\quad(j\in J_0).
\]
归一化条件与目标均不变，故完整对偶也取得最优值。

\textit{原始最优解与正性。}
可行点 $(a,u,\lambda,w)=(\ell(0)\mathbf1,0,0,0)$ 的目标为 $C=2\ell(0)$。
在 $J\le C$ 上，
\[
0\le a_i\le C/q_i,\qquad
0\le w_{kj}\le\lambda_k\le C/\theta_k,\qquad
\ell_k(u_r)\le C/q_i+(C/\theta_k)\sum_j\Delta_{ir,j}.
\]
与有限问题相同，这些界给出包含最优解的紧致限制（常数损失时令 $u=0$）。
若最优值为零，则 $a=\lambda=w=0$，从而
\[
0<2\ell(0)\le\ell(-u_r)+\ell(u_r)\le0,
\]
矛盾。
\end{proof}

\begin{proposition}[仿射对偶与支撑割]\label{prop:dual}
对于每个 $z\in[0,1]^D$，
\begin{equation}\label{eq:appendix-affine}
F(z)=\max_{(\Pi,\alpha,\mu)\in\mathcal D}
\{2\sqrt{H(\Pi)}-\alpha-\mu^\top z\}.
\end{equation}
若 $(\bar\Pi,\bar\alpha,\bar\mu)$ 为 $z$ 处的归一化对偶最优解，
则一个仿射对偶最优解为
\begin{equation}\label{eq:dual-rescaling}
(\Pi,\alpha,\mu)=F(z)(\bar\Pi,\bar\alpha,\bar\mu),
\qquad H(\Pi)=F(z)^2.
\end{equation}
从而 $F$ 为凸函数。对于 $z^t$ 处的仿射最优解
$(\Pi^t,\alpha^t,\mu^t)$，
\[
-\mu^t\in\partial F(z^t),\qquad
F(z)\ge F(z^t)-(\mu^t)^\top(z-z^t)\quad(z\in[0,1]^D).
\]
这里的次梯度不等式定义在单位立方体上。
\end{proposition}
\begin{proof}
$\mathcal D$ 为锥，且对 $\tau\ge0$ 有 $H(\tau\Pi)=\tau H(\Pi)\ge0$。
对 $(\Pi,\alpha,\mu)\in\mathcal D$，令 $b=\alpha+\mu^\top z$。
若 $b>0$，除以 $b$ 归一化得 $H(\Pi)/b\le F(z)$，故
\begin{equation}\label{eq:scaling}
2\sqrt{H(\Pi)}-b
=\frac{H(\Pi)}b
-\left(\sqrt{\frac{H(\Pi)}b}-\sqrt b\right)^2
\le F(z).
\end{equation}
若 $b=0$，非负性与行和等式给出 $\alpha=0$、$\Pi=0$，故仿射值为零。
反之，归一化最优解满足 $H(\bar\Pi)=F(z)>0$，且
\[
H(F(z)\bar\Pi)=F(z)^2,\qquad
F(z)(\bar\alpha+\bar\mu^\top z)=F(z),\qquad
2\sqrt{H(F(z)\bar\Pi)}-F(z)=F(z).
\]
因此所述缩放达到仿射最大值。对 $z^t$ 处的仿射最优解，
\[
F(z)\ge2\sqrt{H(\Pi^t)}-\alpha^t-(\mu^t)^\top z
=F(z^t)-(\mu^t)^\top(z-z^t).
\]
仿射函数的上确界为凸函数，上式给出次梯度 $-\mu^t=-F(z^t)\bar\mu^t$。
$H$ 是线性函数族的非负下确界，故 $H$、$\sqrt H$ 均为凹函数，
固定 $z$ 的对偶为凸优化问题。
\end{proof}

\section{风险保证与搜索代价}\label{app:convergence}
令
\[
\mathcal Z_d=\{\mathbf1_S:S\subseteq[D],\ |S|\le d\},\qquad
\mathrm{OPT}=\min_{z\in\mathcal Z_d}F(z),\qquad F(\mathbf1_S)=V(S).
\]
在 $z^s$，设 $p_s$ 为原始可行值，$f_s$ 为其恢复评分函数，$c_s$ 为对偶可行仿射割：
\[
R(f_s)\le p_s,\qquad c_s(z)\le F(z)\quad(z\in[0,1]^D).
\]
割使用精确的 $H(\Pi)$ 或经认证的下界 $0\le h\le H(\Pi)$。
代入候选评分只给出该下确界的上界，不能认证下界割。

\begin{proposition}[非精确求解的保证]\label{prop:inexact}
在完成 $t$ 次评估后，定义
\[
U_t=\min_{s\le t}p_s,\qquad
m_t(z)=\max\{0,c_1(z),\ldots,c_t(z)\}.
\]
设 $(z^{t+1},\eta^{t+1})$ 为任意主问题可行解，满足
$\eta^{t+1}\ge m_t(z^{t+1})$；
设 $L_t$ 为 $\min_{z\in\mathcal Z_d}m_t(z)$ 的有效全局下界。
若 $f_{\rm inc}$ 为最小 $p_s$ 对应的评分函数，则
\begin{equation}\label{eq:inexact-certificate}
L_t\le\mathrm{OPT}\le R(f_{\rm inc})\le U_t,\qquad
0\le R(f_{\rm inc})-\mathrm{OPT}\le U_t-L_t.
\end{equation}
对于 $\delta,\beta\ge0$ 和 $\varepsilon>0$，若每个已评估点均满足
\[
p_s-c_s(z^s)\le\delta,\qquad
\eta^{t+1}-L_t\le\beta,\qquad
\delta+\beta\le\varepsilon,
\]
则重复选点 $z^{t+1}=z^s$ 必有 $U_t-L_t\le\varepsilon$。
因此，只要每次迭代都满足这些精度，
以 $U_t-L_t\le\varepsilon$ 为停止条件的实现至多进行 $|\mathcal Z_d|$ 次评估。
\end{proposition}
\begin{proof}
由割的有效性与原始可行性，
\[
0\le m_t\le F,\qquad
L_t\le\min_{\mathcal Z_d}m_t\le\mathrm{OPT}
\le R(f_{\rm inc})\le U_t.
\]
若 $z^{t+1}=z^s$，则 $U_t\le p_s$、$\eta^{t+1}\ge c_s(z^s)$，故
\[
U_t-L_t\le p_s-c_s(z^s)+\eta^{t+1}-L_t
\le\delta+\beta\le\varepsilon.
\]
故每次未终止的迭代均选择 $\mathcal Z_d$ 中的新点。
风险下界须使用全局主问题界 $L_t$；仅有可行主问题目标 $\eta^{t+1}$ 不足以保证该结论。
\end{proof}

\begin{proposition}[依赖数据的斜率界]\label{prop:bounded}
与扩展模型一致，设 $\Delta_{ir,j}=|x_{ij}-x_{rj}|$，
$q_i>0$，$\sum_{i\in I_k}q_i=1$，$\theta_k>0$。令
\begin{equation}\label{eq:slope-bound}
C=2\ell(0),\qquad
A_{kj}=\sum_{i\in I_k}q_i\max_{r\in[N]}\Delta_{ir,j},\qquad
M_j=C\max_{k=0,1}\frac{A_{kj}}{\theta_k}.
\end{equation}
在每个 $z\in[0,1]^D$，均存在满足每个坐标 $j$ 上
$0\le\mu_j\le M_j$ 的仿射对偶最优解。因此
\[
|F(z)-F(z')|\le\sum_jM_j|z_j-z_j'|
\qquad(z,z'\in[0,1]^D).
\]
\end{proposition}
\begin{proof}
对命题~\ref{prop:dual} 的缩放最优解，由 $0\le\bar\alpha\le1$ 得
$0\le\alpha=F(z)\bar\alpha\le F(z)\le C$。行和等式给出
$d_{kj}(\Pi)=\sum_{i\in I_k,r}\Pi_{ir}\Delta_{ir,j}\le\alpha A_{kj}$.
令 $g_j=\max_k d_{kj}(\Pi)/\theta_k$、$\mu_j'=\min\{\mu_j,g_j\}$。
截断坐标满足 $(d_{kj}(\Pi)-\theta_k\mu_j')_+=0$，其余坐标不变。
因此 $(\Pi,\alpha,\mu')\in\mathcal D$，且
\[
F(z)\ge2\sqrt{H(\Pi)}-\alpha-(\mu')^\top z
\ge2\sqrt{H(\Pi)}-\alpha-\mu^\top z=F(z),\qquad
0\le\mu_j'\le g_j\le\alpha\max_k A_{kj}/\theta_k\le M_j.
\]
分别在 $z$、$z'$ 使用上述乘子构造支撑割，得
\[
-\sum_jM_j|z_j-z_j'|\le F(z)-F(z')\le\sum_jM_j|z_j-z_j'|.
\]
\end{proof}

\begin{proposition}[精确搜索与选点间隔]\label{prop:convergence}
设所有固定选集问题与主问题均精确求解。
每次评估均选取命题~\ref{prop:bounded} 中的有界仿射最优解，
并在 $U_t-\eta^{t+1}\le\varepsilon$ 时停止，
其中 $U_t=\min_{s\le t}F(z^s)$。
对于返回的评分函数 $f^*$ 与最后一次迭代 $T$，
\begin{equation}\label{eq:exact-certificate}
0\le R(f^*)-\mathrm{OPT}\le U_T-\eta^{T+1}\le\varepsilon.
\end{equation}
对于任意后评估选集 $S_t$ 与先评估选集 $S_s$，
\begin{equation}\label{eq:selection-separation}
\sum_{j\in S_t\setminus S_s}M_j>\varepsilon,\qquad s<t.
\end{equation}
定义加权伪距离 $d_M(S,T)=\sum_{j\in S\triangle T}M_j$，
以及有限选集上的分离集大小
\[
\mathcal P_\varepsilon=
\max\{|\mathcal A|:\mathcal A\subseteq\{S:|S|\le d\},\
d_M(S,T)>\varepsilon\text{，对不同的 }S,T\in\mathcal A\}.
\]
允许 $M_j=0$，有
\begin{equation}\label{eq:search-cost}
T\le\mathcal P_\varepsilon\le\sum_{r=0}^d\binom Dr,\qquad
\#\mathrm{oracle}=T,\quad
\#\mathrm{master}=T,\quad
\#\mathrm{recovery}=1.
\end{equation}
这一依赖数据的分离集上界未必是多项式界。
\end{proposition}
\begin{proof}
在命题~\ref{prop:inexact} 中取 $p_s=c_s(z^s)=F(z^s)$、
$\delta=\beta=0$、$L_t=\eta^{t+1}$。
恢复的最优检验风险为 $U_T$，即得风险保证。
对 $s<t$，在评估 $S_t$ 之前尚未终止，故
\[
\begin{aligned}
c_s(\mathbf1_{S_t})&\le\eta^t<U_{t-1}-\varepsilon
\le F(\mathbf1_{S_s})-\varepsilon,\\
\varepsilon&<(\mu^s)^\top(\mathbf1_{S_t}-\mathbf1_{S_s})
=\sum_{j\in S_t\setminus S_s}\mu_j^s
-\sum_{j\in S_s\setminus S_t}\mu_j^s
\le\sum_{j\in S_t\setminus S_s}M_j\le d_M(S_s,S_t).
\end{aligned}
\]
已评估选集构成 $\varepsilon$ 分离集，故 $T\le\mathcal P_\varepsilon$。
算法~\ref{alg:oa} 直接给出求解次数。
第 $t$ 个主问题有 $D$ 个二元变量、一个连续变量及 $t$ 个割；这些计数不限制单次求解时间。
\end{proof}
