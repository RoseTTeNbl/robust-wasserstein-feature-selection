\begin{abstract}
本文推导带变量选择的 Wasserstein 鲁棒检验问题的精确有限维凸表示。利用统一的风险上界引入选择变量，保持二元点处的最优值不变。所得凸值函数具有仿射对偶割，可用于在变量选集上优化。
\end{abstract}


\section{问题表述}\label{sec:problem}

本文研究鲁棒假设检验中的变量选择问题。令 $[D]=\{1,\ldots,D\}$，以正整数 $d$ 表示允许选择的变量数。
给定检测函数 $f:~\mathbb{R}^D\to\mathbb{R}$，遵循文献 \cite{gao2018robust} 的约定，将最坏情形风险定义为


\begin{equation}\label{eq:risk}
R(f)=\sup_{\mathbb{P}_0\in\mathcal{P}_0, \mathbb{P}_1\in\mathcal{P}_1}~
\left\{\mathbb E_{\mathbb P_0}\ell(-f(X))+
       \mathbb E_{\mathbb P_1}\ell(f(X))\right\},
\end{equation}
其中，$\mathcal{P}_k, k=0,1$ 为基于 Wasserstein 距离构造的分布不确定集；生成函数 $\ell(\cdot)$ 非负、非减且为凸函数，并满足 $\ell(0)=1, \lim_{t\to-\infty}\ell(t)=0$。
对于 $k=0,1$，假设第 $k$ 类中有 $n_k$ 个观测，记为 $\{x_i^{(k)}\}_{i=1}^{n_k}$。
将相应的经验分布记为 $\widehat{\mathbb{P}}_k=\frac1{n_k}\sum_{i=1}^{n_k}\delta_{x_i^{(k)}}$，构造 Wasserstein 不确定集
\[
\mathcal{P}_k=\Big\{
\mathbb{P}_k:~\mathcal{W}_1(
\widehat{\mathbb{P}}_k, \mathbb{P}_k
)\le \rho_k
\Big\}.
\]
这里，$\mathcal{W}_1(\cdot,\cdot)$ 表示以 $\ell_1$ 范数为运输成本的 $1$-Wasserstein 距离。


对于 $S\subseteq[D]$，定义仅依赖于 $S$ 中坐标的检测函数类：
\[
\mathcal{F}_S = \Big\{
f:~\mathbb{R}^D\to\mathbb{R}, f(x) = g(x[S])\text{，其中 $g:~\mathbb{R}^{|S|}\to\mathbb{R}$ 为某个可测函数}
\Big\}.
\]
对于向量 $x\in\mathbb{R}^D$，以 $x[i]$ 表示其第 $i$ 个坐标，以 $x[S]$ 表示下标属于 $S$ 的坐标组成的向量。
所研究的变量选择问题可写成如下组合优化问题：
\begin{equation}
\min_{S\subseteq[D], |S|\le d}\min_{f\in\mathcal{F}_S}~R(f).
\end{equation}

\section{有限维等价表示}\label{sec:finite}
假设 $n_0,n_1\ge1$、$\rho_0,\rho_1>0$，且 $\ell$ 在 $\mathbb R$ 上有限。
所考虑分布的一阶矩均有限。以下以 $i\in[n_0]$ 表示第 $0$ 类观测，
$j\in[n_1]$ 表示第 $1$ 类观测，$m\in[D]$ 表示一个坐标。对于固定选集，定义
\[
V(S)=\min_{f\in\mathcal F_S}R(f),\qquad
c_{ij}(S)=\|x_i^{(0)}[S]-x_j^{(1)}[S]\|_1.
\]
文献 \cite{gao2018robust} 的最不利分布观点提示我们考察观测之间的经验质量运输。
针对第~\ref{sec:problem} 节的生成损失，下面直接推导有限维问题。
为第 $k$ 类的每个观测 $r\in[n_k]$ 引入辅助评分 $u_r^{(k)}$ 和源损失上界 $a_r^{(k)}$；
以 $\lambda_k\ge0$ 表示第 $k$ 类运输预算的价格。记
\[
J(a,\lambda)=\frac1{n_0}\sum_i a_i^{(0)}+
\frac1{n_1}\sum_j a_j^{(1)}+\rho_0\lambda_0+\rho_1\lambda_1.
\]

\begin{proposition}[有限维等价表示]\label{prop:finite}
对于每个 $S\subseteq[D]$，以下凸优化问题取得最优值 $V(S)$：
\begin{equation}\label{eq:finite}
\begin{aligned}
\min_{a,u,\lambda}\quad&J(a,\lambda)\\
\text{约束为}\quad&\ell(-u_i^{(0)})\le a_i^{(0)},&&i\in[n_0],\\
&\ell(u_j^{(1)})\le a_j^{(1)},&&j\in[n_1],\\
&\ell(-u_j^{(1)})\le a_i^{(0)}+\lambda_0c_{ij}(S),
&&i\in[n_0],\ j\in[n_1],\\
&\ell(u_i^{(0)})\le a_j^{(1)}+\lambda_1c_{ij}(S),
&&i\in[n_0],\ j\in[n_1],\\
&a^{(k)},u^{(k)}\in\mathbb R^{n_k},\quad\lambda_k\ge0,&&k=0,1.
\end{aligned}
\end{equation}
由一个最优解可以构造定义在整个观测空间上的可测检测函数，其风险为 $V(S)$。
\end{proposition}

前两个约束控制留在源观测处的质量所产生的损失，后两个约束控制向另一类的运输。
例如，第 $0$ 类的质量从 $x_i^{(0)}[S]$ 运到 $x_j^{(1)}[S]$ 时，
损失为 $\ell(-u_j^{(1)})$，由源项 $a_i^{(0)}$ 与按价格计入的成本
$\lambda_0c_{ij}(S)$ 之和控制。附录~\ref{app:finite} 通过两个方向证明精确性：
受限运输给出原风险的下界，而每个有限维可行解均可构造风险不超过其目标值的检测函数。
所选坐标上的投影重合时，这一结论仍成立。

零检测函数的风险为 $2$。对角约束蕴含 $a^{(k)}\ge0$，因此最优解满足
\begin{equation}\label{eq:uniform-bound}
\rho_0\lambda_0+\rho_1\lambda_1\le J(a,\lambda)=V(S)\le2.
\end{equation}
这一上界使我们可以用二元变量控制是否使用一个坐标。

\section{选择变量与对偶}\label{sec:selection}
令 $z=\mathbf1_S$，并定义
\[
\mathcal Z_d=\{z\in\{0,1\}^D:\mathbf1^\top z\le d\},\qquad
\Delta_{ij,m}=|x_i^{(0)}[m]-x_j^{(1)}[m]|.
\]
于是 $c_{ij}(S)=\sum_m\Delta_{ij,m}z_m$。
引入 $w_{km}$ 以替代乘积 $\lambda_kz_m$。
约束 $0\le w_{km}\le\lambda_k$ 与
$\rho_0w_{0m}+\rho_1w_{1m}\le2z_m$ 保证当 $z_m=0$ 时，$w_{0m}=w_{1m}=0$。
当 $z_m=1$ 时，由式~\eqref{eq:uniform-bound}，在最优解处可以取
$w_{km}=\lambda_k$。对于 $z\in[0,1]^D$，定义
\begin{equation}\label{eq:value}
\begin{aligned}
F(z)=\min_{a,u,\lambda,w}\quad&J(a,\lambda)\\
\text{约束为}\quad&
\ell(-u_i^{(0)})\le a_i^{(0)},\quad
\ell(u_j^{(1)})\le a_j^{(1)},\\
&\ell(-u_j^{(1)})\le a_i^{(0)}+\sum_m\Delta_{ij,m}w_{0m},\\
&\ell(u_i^{(0)})\le a_j^{(1)}+\sum_m\Delta_{ij,m}w_{1m},\\
&0\le w_{km}\le\lambda_k,\qquad k=0,1,\ m\in[D],\\
&\rho_0w_{0m}+\rho_1w_{1m}\le2z_m,\qquad m\in[D],\\
&a^{(k)},u^{(k)}\in\mathbb R^{n_k},\quad\lambda_k\ge0,
\qquad k=0,1.
\end{aligned}
\end{equation}
损失约束对每个 $i\in[n_0],j\in[n_1]$ 成立。
该模型具有文献 \cite{bertsimas2021unified} 中问题~(2.1)--(2.2) 的两阶段逻辑结构：
每个坐标激活凸连续问题中的变量对 $(w_{0m},w_{1m})$，外层问题至多选择 $d$ 个变量对。
上界 $2$ 由风险导出，无需人为调节。

\begin{proposition}[选择变量表示]\label{prop:selection}
式~\eqref{eq:value} 的最小值可以取得。函数 $F$ 在 $[0,1]^D$ 上为凸函数，且
\begin{equation}\label{eq:binary-exact}
F(\mathbf1_S)=V(S),\qquad
\min_{|S|\le d}V(S)=\min_{z\in\mathcal Z_d}F(z).
\end{equation}
\end{proposition}
\begin{proof}
式~\eqref{eq:finite} 的最优解可以扩展为式~\eqref{eq:value} 的可行解：
在 $z=\mathbf1_S$ 处令 $w_{km}=\lambda_kz_m$，可行性由式~\eqref{eq:uniform-bound} 保证。
反过来，该二元点处的可行性蕴含 $S$ 之外有 $w_{km}=0$，且
\[
\sum_m\Delta_{ij,m}w_{km}\le
\lambda_k\sum_{m\in S}\Delta_{ij,m}=\lambda_kc_{ij}(S).
\]
因此，该解对式~\eqref{eq:finite} 可行，目标值不变，从而二元点处的最优值相等。
对于任意 $z$，零检测函数给出值为 $2$ 的可行目标。
在 $J\le2$ 上，源损失上界和价格均有界；由附录~\ref{app:finite} 的论证，
每个评分的正负两侧也均有界，且 $0\le w_{km}\le\lambda_k$。故最优解存在。
式~\eqref{eq:value} 的所有约束关于包括 $z$ 在内的变量联合凸。
$z,z'$ 处最优解的凸组合在 $tz+(1-t)z'$ 处可行，故
\[
F(tz+(1-t)z')\le tF(z)+(1-t)F(z'),\qquad 0\le t\le1.
\]
\end{proof}

\subsection{对偶表示}\label{sec:dual}
令 $\Gamma_{ij}^{(0)}\ge0$ 为第 $0$ 类跨类约束的乘子，
$\Gamma_{ij}^{(1)}\ge0$ 为第 $1$ 类跨类约束的乘子。
它们分别表示从源观测 $i$ 运到目标观测 $j$、从源观测 $j$ 运到目标观测 $i$ 的质量。
记 $\Gamma=(\Gamma^{(0)},\Gamma^{(1)})$。
在两类质量为 $p,q\ge0$ 的位置，最优加权损失为
\[
h(p,q)=\inf_{v\in\mathbb R}\{p\ell(-v)+q\ell(v)\}.
\]
在第 $0$ 类观测处，保留的第 $0$ 类质量为 $1/n_0-\sum_j\Gamma_{ij}^{(0)}$，
运入的第 $1$ 类质量为 $\sum_j\Gamma_{ij}^{(1)}$。
对第 $1$ 类作相同分析，得到优化后的总损失
\begin{equation}\label{eq:dual-loss}
\begin{aligned}
H(\Gamma)={}&\sum_i
h\left(\frac1{n_0}-\sum_j\Gamma_{ij}^{(0)},\sum_j\Gamma_{ij}^{(1)}\right)\\
&+\sum_j
h\left(\sum_i\Gamma_{ij}^{(0)},\frac1{n_1}-\sum_i\Gamma_{ij}^{(1)}\right).
\end{aligned}
\end{equation}
令 $\mu_m\ge0$ 为坐标约束 $\rho_0w_{0m}+\rho_1w_{1m}\le2z_m$ 的乘子，并定义
\begin{equation}\label{eq:dual-domain}
\mathcal D=\left\{(\Gamma,\mu)\ge0:
\begin{aligned}
&\sum_j\Gamma_{ij}^{(0)}\le1/n_0&&(i\in[n_0]),\\
&\sum_i\Gamma_{ij}^{(1)}\le1/n_1&&(j\in[n_1]),\\
&\sum_m\left(\sum_{i,j}\Gamma_{ij}^{(k)}\Delta_{ij,m}
-\rho_k\mu_m\right)_+\le\rho_k&&(k=0,1)
\end{aligned}\right\},
\end{equation}
其中 $(v)_+=\max\{v,0\}$，各求和均使用前文定义的下标范围。
前两个约束限制运出的质量；后两个约束由拉格朗日函数对 $\lambda,w$ 最小化得到，
推导见附录~\ref{app:dual}。集合 $\mathcal D$ 与 $z$ 无关。

\begin{proposition}[对偶表示]\label{prop:dual}
对于每个 $z\in[0,1]^D$，以下最大值可以取得：
\begin{equation}\label{eq:dual}
F(z)=\max_{(\Gamma,\mu)\in\mathcal D}
\{H(\Gamma)-2\mu^\top z\}.
\end{equation}
$z^t$ 处的最优解 $(\Gamma^t,\mu^t)$ 给出支撑函数
\begin{equation}\label{eq:cut}
\begin{aligned}
c_t(z)&=H(\Gamma^t)-2(\mu^t)^\top z\\
&=F(z^t)-2(\mu^t)^\top(z-z^t)\le F(z),
\end{aligned}
\end{equation}
并满足 $c_t(z^t)=F(z^t)$。
\end{proposition}

函数 $h$ 关于权重为凹函数，因此 $H$ 为凹函数，而 $\mathcal D$ 为凸集。
所以固定 $z$ 时，对偶是凸优化问题。它关于 $z$ 的仿射形式给出了
文献 \cite{bertsimas2021unified} 中定理~2.5 和算法~3.1 所依据的切平面构造。
本模型的容量约束在附录~\ref{app:dual} 中直接处理。

\section{切平面算法}\label{sec:algorithm}
记 $\mathrm{OPT}=\min_{z\in\mathcal Z_d}F(z)$。
评估 $z^1,\ldots,z^t$ 并取得相应的对偶最优解后，
令 $U_t=\min_{s\le t}F(z^s)$，并求解
\begin{equation}\label{eq:master}
\begin{aligned}
(z^{t+1},L_t)\in\operatorname*{arg\,min}_{z\in\mathcal Z_d,\,\eta\ge0}
\quad&\eta\\
\text{约束为}\quad&
\eta\ge F(z^s)-2(\mu^s)^\top(z-z^s),\qquad s=1,\ldots,t.
\end{aligned}
\end{equation}
每条保留的割均为全局下界，并在生成该割的已评估选集处取等号。因此，
\begin{equation}\label{eq:bounds}
L_t\le\mathrm{OPT}\le U_t.
\end{equation}
从任意 $z^1\in\mathcal Z_d$ 出发，交替求解凸子问题
\eqref{eq:value}--\eqref{eq:dual} 与二元线性主问题~\eqref{eq:master}。
当 $U_t-L_t\le\varepsilon$ 时停止，并返回最好的已评估选集。
由式~\eqref{eq:extension} 恢复的检测函数具有风险 $U_t$，与最优风险的差距至多为
$\varepsilon$。若内层求解非精确，使用实际可行对偶表达式
$H(\Gamma^t)-2(\mu^t)^\top z$ 作为割，以原始可行解的目标值作为上界；
主问题必须提供全局下界。该表示在二元点处精确，但其连续松弛可能较弱。

\section{有限维等价表示的证明}\label{app:finite}
\begin{proof}[命题~\ref{prop:finite} 的证明]
以 $W(S)$ 表示式~\eqref{eq:finite} 的下确界。对于 $f(x)=g(x[S])$，令
$u_r^{(k)}=g(x_r^{(k)}[S])$。限制运输方式为：将质量
$s_i^{(0)},s_j^{(1)}\ge0$ 留在源观测处，或将质量
$\Gamma_{ij}^{(0)},\Gamma_{ij}^{(1)}\ge0$ 运到另一类观测的所选坐标处，
方向与第~\ref{sec:dual} 节相同：
\begin{equation}\label{eq:restricted-transport}
\begin{gathered}
s_i^{(0)}+\sum_j\Gamma_{ij}^{(0)}=1/n_0,\qquad
s_j^{(1)}+\sum_i\Gamma_{ij}^{(1)}=1/n_1,\\
\sum_{i,j}c_{ij}(S)\Gamma_{ij}^{(k)}\le\rho_k,
\qquad k=0,1.
\end{gathered}
\end{equation}
第 $0$ 类的目标为
\[
\sum_i s_i^{(0)}\ell(-u_i^{(0)})
+\sum_{i,j}\Gamma_{ij}^{(0)}\ell(-u_j^{(1)}).
\]
取自由质量乘子 $a_i^{(0)}$ 和预算乘子 $\lambda_0\ge0$，
对应最大化问题的拉格朗日函数为
\begin{equation}\label{eq:first-lagrangian}
\begin{aligned}
&\frac1{n_0}\sum_i a_i^{(0)}+\rho_0\lambda_0
+\sum_i s_i^{(0)}[\ell(-u_i^{(0)})-a_i^{(0)}]\\
&\hspace{12pt}
+\sum_{i,j}\Gamma_{ij}^{(0)}
[\ell(-u_j^{(1)})-a_i^{(0)}-\lambda_0c_{ij}(S)].
\end{aligned}
\end{equation}
其上确界有限当且仅当两组方括号中的系数均非正。
第 $1$ 类的计算给出式~\eqref{eq:finite} 中的另外两个约束。
两个运输线性规划均可行且有界；由线性规划对偶性，其最大风险等于
固定这些评分时 $J(a,\lambda)$ 的最小值。
仅将源观测的所选坐标替换为目标坐标，即可将每个运输计划扩展到原不确定集，
成本为 $c_{ij}(S)$，损失不变。再对 $u$ 最小化，得到 $W(S)\le R(f)$，
从而 $W(S)\le V(S)$。

反过来，对于可行解 $(a,u,\lambda)$，定义
\[
E_k(y)=\min_{r\in[n_k]}
\{a_r^{(k)}+\lambda_k\|x_r^{(k)}[S]-y\|_1\},\qquad k=0,1,
\]
其中 $y\in\mathbb R^{|S|}$。分别选择达到 $E_0(y)$、$E_1(y)$ 的最小下标
$i=i(y)$、$j=j(y)$，并令
\[
r_0=\|x_i^{(0)}[S]-y\|_1,\qquad
r_1=\|x_j^{(1)}[S]-y\|_1,\qquad b=c_{ij}(S).
\]
定义
\begin{equation}\label{eq:extension}
g(y)=\begin{cases}
(1-\gamma)u_i^{(0)}+\gamma u_j^{(1)},\quad
\gamma=\min\{1,r_0/b\},&b>0,\\
u_i^{(0)},&b=0.
\end{cases}
\end{equation}
当 $b>0$ 时，三角不等式给出 $\gamma b\le r_0$ 和 $(1-\gamma)b\le r_1$。
由凸性和四个约束，得到
\begin{equation}\label{eq:envelopes}
\begin{aligned}
\ell(-g(y))
&\le(1-\gamma)\ell(-u_i^{(0)})+\gamma\ell(-u_j^{(1)})\\
&\le a_i^{(0)}+\lambda_0\gamma b\le E_0(y),\\
\ell(g(y))
&\le(1-\gamma)\ell(u_i^{(0)})+\gamma\ell(u_j^{(1)})\\
&\le a_j^{(1)}+\lambda_1(1-\gamma)b\le E_1(y).
\end{aligned}
\end{equation}
当 $b=0$ 时，第 $0$ 类的对角约束与第 $1$ 类的跨类约束给出相同上界。
有限个下标中的确定性择取规则保证 $g$ 为 Borel 可测函数，且
$|g|\le\max_{k,r}|u_r^{(k)}|$。对于完整空间中的每个输入 $x$ 和源观测，
\[
\begin{aligned}
\ell(-g(x[S]))&\le a_i^{(0)}+
\lambda_0\|x_i^{(0)}[S]-x[S]\|_1
\le a_i^{(0)}+\lambda_0\|x_i^{(0)}-x\|_1,\\
\ell(g(x[S]))&\le a_j^{(1)}+
\lambda_1\|x_j^{(1)}[S]-x[S]\|_1
\le a_j^{(1)}+\lambda_1\|x_j^{(1)}-x\|_1.
\end{aligned}
\]
对于 $\delta>0$ 和任意 $\mathbb P_k\in\mathcal P_k$，
对从经验分布到 $\mathbb P_k$、成本至多为 $\rho_k+\delta$ 的耦合积分：
\[
\begin{aligned}
\mathbb E_{\mathbb P_0}\ell(-g(X[S]))
&\le\frac1{n_0}\sum_i a_i^{(0)}+\lambda_0(\rho_0+\delta),\\
\mathbb E_{\mathbb P_1}\ell(g(X[S]))
&\le\frac1{n_1}\sum_j a_j^{(1)}+\lambda_1(\rho_1+\delta).
\end{aligned}
\]
令 $\delta\downarrow0$ 并取 $f(x)=g(x[S])$，得到 $R(f)\le J(a,\lambda)$。
因此 $V(S)\le W(S)$，等价性得证。

可行点 $a^{(k)}=\mathbf1$、$u^{(k)}=0$、$\lambda=0$ 的目标值为 $2$。
在 $J\le2$ 上，有 $0\le a_r^{(k)}\le2n_k$ 和 $0\le\lambda_k\le2/\rho_k$。
每个评分的两类损失分别由对角约束和异类约束控制。
由于当 $t\to\infty$ 时有 $\ell(t)\to\infty$，所有评分均有界。
该子水平集紧致，故最优解存在；恢复出的检测函数取得 $V(S)$。
\end{proof}

\section{对偶表示的证明}\label{app:dual}
\begin{proof}[命题~\ref{prop:dual} 的证明]
除第~\ref{sec:dual} 节的 $\Gamma,\mu$ 外，为对角损失约束分配非负乘子
$s_i^{(0)},s_j^{(1)}$，为 $w_{km}\le\lambda_k$ 分配非负乘子 $\beta_{km}$。
保留 $a,u$ 为自由变量、$\lambda,w\ge0$，最小化问题的拉格朗日函数为
\begin{equation}\label{eq:full-lagrangian}
\begin{aligned}
\mathcal L={}&J(a,\lambda)
+\sum_i s_i^{(0)}[\ell(-u_i^{(0)})-a_i^{(0)}]
+\sum_j s_j^{(1)}[\ell(u_j^{(1)})-a_j^{(1)}]\\
&+\sum_{i,j}\Gamma_{ij}^{(0)}
\left[\ell(-u_j^{(1)})-a_i^{(0)}-\sum_m\Delta_{ij,m}w_{0m}\right]\\
&+\sum_{i,j}\Gamma_{ij}^{(1)}
\left[\ell(u_i^{(0)})-a_j^{(1)}-\sum_m\Delta_{ij,m}w_{1m}\right]\\
&+\sum_{k,m}\beta_{km}(w_{km}-\lambda_k)
+\sum_m\mu_m(\rho_0w_{0m}+\rho_1w_{1m}-2z_m).
\end{aligned}
\end{equation}
对 $a$ 取下确界时，其值有限当且仅当
\[
s_i^{(0)}+\sum_j\Gamma_{ij}^{(0)}=1/n_0,\qquad
s_j^{(1)}+\sum_i\Gamma_{ij}^{(1)}=1/n_1.
\]
消去 $s\ge0$ 得到式~\eqref{eq:dual-domain} 的前两个约束。
与评分有关的项为
\[
\begin{aligned}
&\sum_i\left[s_i^{(0)}\ell(-u_i^{(0)})+
\left(\sum_j\Gamma_{ij}^{(1)}\right)\ell(u_i^{(0)})\right]
\\&\quad+\sum_j\left[\left(\sum_i\Gamma_{ij}^{(0)}\right)\ell(-u_j^{(1)})+
s_j^{(1)}\ell(u_j^{(1)})\right],
\end{aligned}
\]
其下确界为式~\eqref{eq:dual-loss} 中的 $H(\Gamma)$。
$\lambda,w\ge0$ 的系数为
\[
\lambda_k:\quad\rho_k-\sum_m\beta_{km},\qquad
w_{km}:\quad\beta_{km}+\rho_k\mu_m-
\sum_{i,j}\Gamma_{ij}^{(k)}\Delta_{ij,m}.
\]
这些系数非负且 $\beta\ge0$，当且仅当
\[
\sum_m\beta_{km}\le\rho_k,\qquad
\beta_{km}\ge
\left(\sum_{i,j}\Gamma_{ij}^{(k)}\Delta_{ij,m}-\rho_k\mu_m\right)_+.
\]
消去 $\beta$ 得到式~\eqref{eq:dual-domain} 的后两个约束，
剩余对偶目标为 $H(\Gamma)-2\mu^\top z$。

固定 $z\in[0,1]^D$，对所有满足 $z_m=0$ 的坐标，消去 $w_{km}=0$ 及其上限和容量约束。
剩余系统具有 Slater 点
\[
u^{(k)}=0,\qquad a^{(k)}=2\mathbf1,\qquad
w_{km}=\frac{z_m}{2\rho_k},\qquad
\lambda_k=1+\frac1{2\rho_k}:
\]
所有损失约束严格成立，且 $0<w_{km}<\lambda_k$、
$\rho_0w_{0m}+\rho_1w_{1m}=z_m<2z_m$。
因此强对偶成立，且对偶最优值可取得。对已消去的坐标，恢复
\[
\mu_m=\max_{k=0,1}\frac1{\rho_k}
\sum_{i,j}\Gamma_{ij}^{(k)}\Delta_{ij,m},\qquad
\beta_{km}=0\quad(z_m=0).
\]
全部系数条件均成立，目标值不变，因此完整对偶取得 $F(z)$。

任意 $(\Gamma^t,\mu^t)\in\mathcal D$ 给出全局下界
$H(\Gamma^t)-2(\mu^t)^\top z\le F(z)$。
在 $z^t$ 处的最优性保证此处取等号，故
\[
F(z)\ge F(z^t)-2(\mu^t)^\top(z-z^t).
\]
\end{proof}
