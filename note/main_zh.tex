\section{问题表述}\label{sec:problem}

本文研究鲁棒假设检验中的变量选择问题。令 $[D]=\{1,\ldots,D\}$，以正整数 $d$ 表示允许选择的变量数。
给定检测函数 $f:~\mathbb{R}^D\to\mathbb{R}$，遵循文献 \cite{gao2018robust} 的约定，将最坏情形风险定义为


\begin{equation}\label{eq:risk}
R(f)=\sup_{\mathbb{P}_0\in\mathcal{P}_0, \mathbb{P}_1\in\mathcal{P}_1}~
\left\{\mathbb E_{\mathbb P_0}\ell(-f(X))+
       \mathbb E_{\mathbb P_1}\ell(f(X))\right\},
\end{equation}
其中，$\mathcal{P}_k, k=0,1$ 为基于 Wasserstein 距离构造的分布不确定集；生成函数 $\ell(\cdot)$ 非负、非减且为凸函数，并满足 $\ell(0)=1, \lim_{t\to-\infty}\ell(t)=0$。
对于 $k=0,1$，假设第 $k$ 类中有 $n_k$ 个观测，记为 $\{x_i^{(k)}\}_{i=1}^{n_k}$。
将相应的经验分布记为 $\widehat{\mathbb{P}}_k=\frac1{n_k}\sum_{i=1}^{n_k}\delta_{x_i^{(k)}}$，构造 Wasserstein 不确定集
\[
\mathcal{P}_k=\Big\{
\mathbb{P}_k:~\mathcal{W}_1(
\widehat{\mathbb{P}}_k, \mathbb{P}_k
)\le \rho_k
\Big\}.
\]
这里，$\mathcal{W}_1(\cdot,\cdot)$ 表示以 $\ell_1$ 范数为运输成本的 $1$-Wasserstein 距离。


对于 $S\subseteq[D]$，定义仅依赖于 $S$ 中坐标的检测函数类：
\[
\mathcal{F}_S = \Big\{
f:~\mathbb{R}^D\to\mathbb{R}, f(x) = g(x[S])\text{，其中 $g:~\mathbb{R}^{|S|}\to\mathbb{R}$ 为某个可测函数}
\Big\}.
\]
对于向量 $x\in\mathbb{R}^D$，以 $x[i]$ 表示其第 $i$ 个坐标，以 $x[S]$ 表示下标属于 $S$ 的坐标组成的向量。
所研究的变量选择问题可写成如下组合优化问题：
\begin{equation}
\min_{S\subseteq[D], |S|\le d}\min_{f\in\mathcal{F}_S}~R(f).
\end{equation}

\section{凸优化表示}\label{sec:finite}
假设 $n_0,n_1\ge1$、$\rho_0,\rho_1>0$，且 $\ell$ 在 $\mathbb R$ 上有限。
所有分布的一阶矩均有限。固定 $S\subseteq[D]$，记
\[
V(S)=\min_{f\in\mathcal F_S}R(f),\qquad
\mathcal X=\{x_i^{(0)}:i\in[n_0]\}\cup
\{x_j^{(1)}:j\in[n_1]\}.
\]
以下引理将文献 \cite[引理~2，式~(7)]{xie2021robust} 的有限支撑表示
推广至第~\ref{sec:problem}~节的生成损失。
证明见附录~\ref{app:recovery}，其中包括投影与检测函数延拓。

\begin{lemma}[有限维最不利分布重构]\label{prop:lfd}
固定选集的风险为
\begin{equation}\label{eq:lfd}
\begin{aligned}
V(S)=\max_{p,\gamma\ge0}\quad&
\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\{p_0(y)\ell(-v)+p_1(y)\ell(v)\}\\
\text{约束为}\quad&
\sum_{y\in\mathcal X}\gamma_{i,y}^{(k)}=\frac1{n_k},
&&i\in[n_k],\ k=0,1,\\
&p_k(y)=\sum_{i=1}^{n_k}\gamma_{i,y}^{(k)},
&&y\in\mathcal X,\ k=0,1,\\
&\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}\|x_i^{(k)}[S]-y[S]\|_1\le\rho_k,
&&k=0,1.
\end{aligned}
\end{equation}
\end{lemma}

其中，$\gamma_{i,y}^{(k)}$ 是从 $x_i^{(k)}[S]$ 运至 $y[S]$ 的质量，
$p_k(y)$ 为目标点质量。

以 $g_y$ 表示赋给 $y[S]$ 的评分。代入目标点质量，得到
\[
\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\{p_0(y)\ell(-v)+p_1(y)\ell(v)\}
=\inf_g\left\{
\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\sum_{j,y}\gamma_{j,y}^{(1)}\ell(g_y)\right\}.
\]
目标函数关于 $g$ 为凸函数、关于 $\gamma$ 为线性函数，运输可行集为紧凸集。
由有限维极小极大定理，得到
\[
V(S)=\inf_g\left\{
\max_{\gamma^{(0)}}\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\max_{\gamma^{(1)}}\sum_{j,y}\gamma_{j,y}^{(1)}\ell(g_y)
\right\},
\]
其中，各最大化问题满足式~\eqref{eq:lfd} 中相应的源质量与运输预算约束。

固定评分向量 $g$，对式~\eqref{eq:lfd} 中的质量约束与运输约束，
分别引入乘子 $a_i^{(0)}\in\mathbb R$ 和 $\lambda_0\ge0$。
关于 $\gamma^{(0)}$ 的最大化问题具有拉格朗日函数
\begin{equation}\label{eq:transport-lagrangian}
\begin{aligned}
\mathcal L_0={}&\sum_{i,y}\gamma_{i,y}^{(0)}\ell(-g_y)
+\sum_i a_i^{(0)}\left(\frac1{n_0}-\sum_y\gamma_{i,y}^{(0)}\right)\\
&+\lambda_0\left(\rho_0-\sum_{i,y}\gamma_{i,y}^{(0)}
\|x_i^{(0)}[S]-y[S]\|_1\right)\\
={}&\frac1{n_0}\sum_i a_i^{(0)}+\rho_0\lambda_0\\
&+\sum_{i,y}\gamma_{i,y}^{(0)}
\left[\ell(-g_y)-a_i^{(0)}
-\lambda_0\|x_i^{(0)}[S]-y[S]\|_1\right].
\end{aligned}
\end{equation}
因此，
\[
\sup_{\gamma^{(0)}\ge0}\mathcal L_0<\infty
\iff
\ell(-g_y)\le a_i^{(0)}+\lambda_0\|x_i^{(0)}[S]-y[S]\|_1,
\quad i\in[n_0],\ y\in\mathcal X.
\]
对两类分别应用线性规划对偶，得到
\begin{equation}\label{eq:finite}
\begin{aligned}
V(S)=\min_{g,a,\lambda}\quad&
\frac1{n_0}\sum_i a_i^{(0)}+\frac1{n_1}\sum_j a_j^{(1)}
+\rho_0\lambda_0+\rho_1\lambda_1\\
\text{约束为}\quad&
\ell(-g_y)\le a_i^{(0)}+\lambda_0\|x_i^{(0)}[S]-y[S]\|_1,
&&i\in[n_0],\ y\in\mathcal X,\\
&\ell(g_y)\le a_j^{(1)}+\lambda_1\|x_j^{(1)}[S]-y[S]\|_1,
&&j\in[n_1],\ y\in\mathcal X,\\
&g\in\mathbb R^{\mathcal X},\quad a^{(k)}\in\mathbb R^{n_k},
\quad\lambda_k\ge0,&&k=0,1.
\end{aligned}
\end{equation}
将该目标函数记为 $J(a,\lambda)$。
附录~\ref{app:recovery} 证明其与可测检测函数上的最优风险等价。

令 $z=\mathbf1_S$，并记
$\mathcal Z_d=\{z\in\{0,1\}^D:\mathbf1^\top z\le d\}$。
运输成本可按坐标分解：
\[
\|x_i^{(k)}[S]-y[S]\|_1
=\sum_{m=1}^D z_m|x_i^{(k)}[m]-y[m]|.
\]
引入 $w_{km}$ 表示坐标是否被激活，并将选择指示变量扩展至
$z\in[0,1]^D$。定义
\begin{equation}\label{eq:value}
\begin{aligned}
F(z)=\min_{g,a,\lambda,w}\quad&J(a,\lambda)\\
\text{约束为}\quad&
\ell(-g_y)\le a_i^{(0)}+\sum_m w_{0m}|x_i^{(0)}[m]-y[m]|,
&&i\in[n_0],\ y\in\mathcal X,\\
&\ell(g_y)\le a_j^{(1)}+\sum_m w_{1m}|x_j^{(1)}[m]-y[m]|,
&&j\in[n_1],\ y\in\mathcal X,\\
&0\le w_{km}\le\lambda_k,&&k=0,1,\ m\in[D],\\
&\rho_0w_{0m}+\rho_1w_{1m}\le2z_m,&&m\in[D],\\
&g\in\mathbb R^{\mathcal X},\quad a^{(k)}\in\mathbb R^{n_k},
\quad\lambda_k\ge0,&&k=0,1.
\end{aligned}
\end{equation}
约束关于 $(z,g,a,\lambda,w)$ 联合凸，故部分最小化定义了
$[0,1]^D$ 上的凸函数 $F$。
当 $z=\mathbf1_S$ 时，约束使 $m\notin S$ 对应的 $w_{km}=0$，
并允许在最优解处对 $m\in S$ 取 $w_{km}=\lambda_k$。
因此，
\begin{equation}\label{eq:binary-exact}
F(\mathbf1_S)=V(S),\qquad
\min_{|S|\le d}V(S)=\min_{z\in\mathcal Z_d}F(z).
\end{equation}
该连续扩展保持每个二元选集处的目标值；在连续预算集合上最小化则得到松弛。
离散问题具有文献 \cite[问题~(2.1)--(2.2)]{bertsimas2021unified} 中逻辑约束模型的形式。
附录~\ref{app:dual} 给出精确性与最优值可达性的证明。

对式~\eqref{eq:value} 的损失约束引入乘子 $\gamma_{i,y}^{(k)}\ge0$，
对坐标约束引入乘子 $\mu_m\ge0$。所得对偶如下。

\begin{proposition}[对偶表示]\label{prop:dual}
对于每个 $z\in[0,1]^D$，
\begin{equation}\label{eq:dual}
F(z)=\max_{(\gamma,\mu)\in\mathcal D}
\{H(\gamma)-2\mu^\top z\},
\end{equation}
其中
\begin{equation}\label{eq:dual-loss}
H(\gamma)=\sum_{y\in\mathcal X}\inf_{v\in\mathbb R}
\left\{\sum_{i=1}^{n_0}\gamma_{i,y}^{(0)}\ell(-v)
+\sum_{j=1}^{n_1}\gamma_{j,y}^{(1)}\ell(v)\right\},
\end{equation}
且
\begin{equation}\label{eq:dual-domain}
\mathcal D=\left\{(\gamma,\mu)\ge0:
\begin{aligned}
&\sum_{y\in\mathcal X}\gamma_{i,y}^{(k)}=1/n_k,
&&i\in[n_k],\ k=0,1,\\
&\sum_{m=1}^D\left(
\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|-\rho_k\mu_m
\right)_+\le\rho_k,
&&k=0,1
\end{aligned}\right\},
\end{equation}
这里，$(v)_+=\max\{v,0\}$，且最大值可以取得。
对于 $z^t$ 处的最优解 $(\gamma^t,\mu^t)$，
\begin{equation}\label{eq:cut}
\begin{aligned}
F(z)&\ge c_t(z):=H(\gamma^t)-2(\mu^t)^\top z\\
&=F(z^t)-2(\mu^t)^\top(z-z^t),\qquad z\in[0,1]^D,
\end{aligned}
\end{equation}
并且 $c_t(z^t)=F(z^t)$。
\end{proposition}
证明见附录~\ref{app:dual}。

\section{外逼近方法}\label{sec:algorithm}
命题~\ref{prop:dual} 给出具有文献
\cite[定理~2.5 与算法~3.1]{bertsimas2021unified} 所述形式的外逼近方法。
令 $\mathrm{OPT}=\min_{z\in\mathcal Z_d}F(z)$。
评估 $z^1,\ldots,z^t$ 并取得相应的对偶最优解后，
令 $U_t=\min_{s\le t}F(z^s)$，求解
\begin{equation}\label{eq:master}
\begin{aligned}
(z^{t+1},L_t)\in\operatorname*{arg\,min}_{z\in\mathcal Z_d,\,\eta\ge0}
\quad&\eta\\
\text{约束为}\quad&
\eta\ge F(z^s)-2(\mu^s)^\top(z-z^s),\qquad s=1,\ldots,t.
\end{aligned}
\end{equation}
由式~\eqref{eq:cut}，
\begin{equation}\label{eq:bounds}
L_t\le\mathrm{OPT}\le U_t.
\end{equation}
从任意 $z^1\in\mathcal Z_d$ 出发，交替求解凸子问题
\eqref{eq:value}--\eqref{eq:dual} 与二元线性主问题~\eqref{eq:master}。
当 $U_t-L_t\le\varepsilon$ 时停止，返回最优的已评估选集。
附录~\ref{app:recovery} 恢复其检测函数，其风险为 $U_t$，最优性间隙至多为 $\varepsilon$。
若内层求解非精确，则以实际可行对偶表达式
$H(\gamma^t)-2(\mu^t)^\top z$ 为割，以原始可行目标值为上界；主问题须提供全局下界。

\section{引理~\ref{prop:lfd} 的证明}\label{app:recovery}
\begin{proof}
下文关于 $\gamma$ 的最大化与关于 $(g,a,\lambda)$ 的最小化
分别受式~\eqref{eq:lfd} 与式~\eqref{eq:finite} 的约束。
令 $S^c=[D]\setminus S$。对于每个源点--目标点对，定义提升后的目标点
\[
\widetilde y_i^{(k)}[S]=y[S],\qquad
\widetilde y_i^{(k)}[S^c]=x_i^{(k)}[S^c],\qquad
\mathbb P_k^\gamma=\sum_{i,y}\gamma_{i,y}^{(k)}
\delta_{\widetilde y_i^{(k)}}.
\]
于是
\[
\mathcal W_1(\widehat{\mathbb P}_k,\mathbb P_k^\gamma)
\le\sum_{i,y}\gamma_{i,y}^{(k)}
\|x_i^{(k)}-\widetilde y_i^{(k)}\|_1
=\sum_{i,y}\gamma_{i,y}^{(k)}
\|x_i^{(k)}[S]-y[S]\|_1\le\rho_k.
\]
因此，对于每个 $f(x)=g(x[S])$，
\begin{equation}\label{eq:finite-lower}
\begin{aligned}
R(f)&\ge\sum_y\{p_0(y)\ell(-g(y[S]))+
p_1(y)\ell(g(y[S]))\}\\
&\ge H(\gamma),
\qquad
\max_\gamma H(\gamma)\le V(S),
\end{aligned}
\end{equation}
其中，$H$ 为式~\eqref{eq:dual-loss} 中的目标函数。
由有限维极小极大定理与式~\eqref{eq:transport-lagrangian}--\eqref{eq:finite}
中的线性规划对偶，得到
\begin{equation}\label{eq:finite-minimax}
\begin{aligned}
\max_\gamma H(\gamma)
&=\inf_{g\in\mathbb R^{\mathcal X}}\max_\gamma
\sum_y\{p_0(y)\ell(-g_y)+p_1(y)\ell(g_y)\}\\
&=\inf_{(g,a,\lambda)}J(a,\lambda).
\end{aligned}
\end{equation}

固定任意可行解 $(g,a,\lambda)$。重合的投影目标点具有相同的约束列，
故可令其评分相等。取 $g(y[S])=g_y$，并定义
\[
E_k(\xi)=\min_{r\in[n_k]}
\{a_r^{(k)}+\lambda_k\|x_r^{(k)}[S]-\xi\|_1\},
\qquad k=0,1.
\]
在投影目标集合之外，分别选取达到 $E_0(\xi),E_1(\xi)$ 的最小下标
$i,j$，令
\[
r_0=\|x_i^{(0)}[S]-\xi\|_1,\quad
r_1=\|x_j^{(1)}[S]-\xi\|_1,\quad
b=\|x_i^{(0)}[S]-x_j^{(1)}[S]\|_1,
\]
\begin{equation}\label{eq:extension}
\tau=
\begin{cases}
\min\{1,r_0/b\},&b>0,\\
0,&b=0,
\end{cases}
\qquad
g(\xi)=(1-\tau)g_{x_i^{(0)}}+\tau g_{x_j^{(1)}}.
\end{equation}
三角不等式给出
\[
b\le r_0+r_1,\qquad
\tau b\le r_0,\qquad (1-\tau)b\le r_1.
\]
由凸性、源点约束及交叉目标点约束，得到
\begin{equation}\label{eq:envelopes}
\begin{aligned}
\ell(-g(\xi))
&\le(1-\tau)\ell(-g_{x_i^{(0)}})+\tau\ell(-g_{x_j^{(1)}})\\
&\le a_i^{(0)}+\lambda_0\tau b
\le a_i^{(0)}+\lambda_0r_0=E_0(\xi),\\
\ell(g(\xi))
&\le(1-\tau)\ell(g_{x_i^{(0)}})+\tau\ell(g_{x_j^{(1)}})\\
&\le a_j^{(1)}+\lambda_1(1-\tau)b
\le a_j^{(1)}+\lambda_1r_1=E_1(\xi).
\end{aligned}
\end{equation}
在投影目标点处，相同的界直接由式~\eqref{eq:finite} 得到。
最小化下标的选取规则为 Borel 可测；赋予有限个观测值仍保持可测性，且
$|g|\le\max_y|g_y|$。
对从经验分布到 $\mathbb P_k$ 的最优耦合积分式~\eqref{eq:envelopes}，
对于每个 $\mathbb P_k\in\mathcal P_k$，得到
\[
\begin{aligned}
\mathbb E_{\mathbb P_0}\ell(-g(X[S]))
&\le\frac1{n_0}\sum_i a_i^{(0)}+\rho_0\lambda_0,\\
\mathbb E_{\mathbb P_1}\ell(g(X[S]))
&\le\frac1{n_1}\sum_j a_j^{(1)}+\rho_1\lambda_1.
\end{aligned}
\]
因此
\[
\max_\gamma H(\gamma)
\le V(S)
\le\inf_{(g,a,\lambda)}J(a,\lambda)
=\max_\gamma H(\gamma).
\]

由于 $g=0$ 时有 $J(\mathbf1,0)=2$，子水平集 $J\le2$ 非空，
其可行点满足
\[
0\le a_i^{(k)}\le2n_k,\qquad
0\le\lambda_k\le2/\rho_k.
\]
损失约束同时控制 $\ell(g_y)$ 与 $\ell(-g_y)$；
由 $t\to\infty$ 时 $\ell(t)\to\infty$，每个 $g_y$ 均有界。
该子水平集紧致，因此有限维最小值及其恢复的检测函数均达到 $V(S)$，
且每个最优解满足
\begin{equation}\label{eq:uniform-bound}
\rho_0\lambda_0+\rho_1\lambda_1\le J(a,\lambda)=V(S)\le2.
\end{equation}
由于 $H$ 在紧运输可行集上为上半连续函数，有限维最不利分布的最大值亦可取得。
\end{proof}

\section{命题~\ref{prop:dual} 的证明}\label{app:dual}
\begin{proof}
由附录~\ref{app:recovery} 中的界及 $0\le w_{km}\le\lambda_k$，
非空子水平集 $J\le2$ 紧致，故 $F(z)$ 取得最小值。
当 $z=\mathbf1_S$ 时，由式~\eqref{eq:uniform-bound}，
固定选集的最优解可通过 $w_{km}=\lambda_kz_m$ 提升。反之，
\[
0\le w_{km}\le\lambda_kz_m
\quad\Longrightarrow\quad
\sum_m w_{km}|x_i^{(k)}[m]-y[m]|
\le\lambda_k\|x_i^{(k)}[S]-y[S]\|_1,
\]
这就证明了式~\eqref{eq:binary-exact}。

对损失约束、$w_{km}\le\lambda_k$ 及坐标容量约束，分别引入非负乘子
$\gamma,\beta,\mu$。保留 $g,a$ 为自由变量、$\lambda,w\ge0$，
整理拉格朗日函数，得到
\begin{equation}\label{eq:full-lagrangian}
\begin{aligned}
\mathcal L={}&-2\mu^\top z
+\sum_{y\in\mathcal X}\left[
\ell(-g_y)\sum_{i=1}^{n_0}\gamma_{i,y}^{(0)}
+\ell(g_y)\sum_{j=1}^{n_1}\gamma_{j,y}^{(1)}\right]\\
&+\sum_{k=0}^1\sum_{i=1}^{n_k}a_i^{(k)}
\left(\frac1{n_k}-\sum_{y\in\mathcal X}\gamma_{i,y}^{(k)}\right)
+\sum_{k=0}^1\lambda_k\left(\rho_k-\sum_m\beta_{km}\right)\\
&+\sum_{k,m}w_{km}\left[
\beta_{km}+\rho_k\mu_m
-\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|\right].
\end{aligned}
\end{equation}
关于 $a$ 的下确界有限，当且仅当 $\sum_y\gamma_{i,y}^{(k)}=1/n_k$；
关于 $g$ 取下确界得到 $H(\gamma)$。
关于 $\lambda,w\ge0$ 的下确界有限，当且仅当
\[
\sum_m\beta_{km}\le\rho_k,\qquad
\beta_{km}\ge
\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|-\rho_k\mu_m.
\]
存在满足上述不等式的非负 $\beta$，当且仅当
\[
\sum_{m=1}^D\left(
\sum_{i=1}^{n_k}\sum_{y\in\mathcal X}
\gamma_{i,y}^{(k)}|x_i^{(k)}[m]-y[m]|-\rho_k\mu_m
\right)_+\le\rho_k,\qquad k=0,1.
\]
因此，对偶可行域与目标函数分别为 $\mathcal D$ 和 $H(\gamma)-2\mu^\top z$。

对于每个 $z\in[0,1]^D$，取
\[
g=0,\qquad \lambda=w=0,\qquad a_i^{(k)}>\ell(0)
\]
满足全部仿射约束，并严格满足全部损失约束。
将非负性计入仿射约束，改进的 Slater 对偶定理给出式~\eqref{eq:dual}
中的等式与对偶最优值的可达性，亦包括 $z_m=0$ 的情形。
因此，$z^t$ 处的最优解 $(\gamma^t,\mu^t)$ 满足
\[
\begin{aligned}
c_t(z)&=H(\gamma^t)-2(\mu^t)^\top z\le F(z),\\
c_t(z^t)&=H(\gamma^t)-2(\mu^t)^\top z^t=F(z^t),\\
c_t(z)&=F(z^t)-2(\mu^t)^\top(z-z^t),
\end{aligned}
\]
这就证明了式~\eqref{eq:cut}。
\end{proof}
